  дія взяти_файл_ю8(компілятор: адреса<Компілятор>, середовище: адреса<Середовище>, шлях: біб::Шлях, вихід_помилки: адреса<адреса<Помилка>>) -> логічне {
    змінна р: ціле;

    змінна взятий_файл: адреса<РозібранийТекст> = пусто;
    якщо знайти_взятий_файл(компілятор.взяті_файли, шлях, взятий_файл::адреса) {
      вернути так;
    }

    змінна код = біб::Інформація { 0, пусто, 0 };

    р = біб::прочитати_файл_за_шляхом(шлях, код::адреса);
    якщо р < 0 {
      вихід_помилки::вміст = створити_помилку(компілятор, "Не вдалось прочитати файл.", пусто);
      вернути ні;
    }

    змінна значення_кд = кд { 0, пусто };
    змінна місцезнаходження_кд = біб::МісцезнаходженняПерекодованого { 0, пусто, 0 };
    якщо не перекодувати_з_ю8_з_місцезнаходженнями(компілятор, ю8 { код.розмір, код.дані }, значення_кд::адреса, місцезнаходження_кд::адреса) {
      вихід_помилки::вміст = створити_помилку(компілятор, "Не вдалось перекодувати файл в кд.", пусто);
      біб::звільнити(код.дані);
      вернути ні;
    }

    змінна оригінальне_значення = біб::виділити<ОригінальнеЗначенняТекстуКоду>(1);
    оригінальне_значення.значення = ю8 { код.розмір, код.дані };
    оригінальне_значення.місцезнаходження = місцезнаходження_кд;

    змінна текст_коду = створити_текст_коду(компілятор);

    р = біб::клонувати_шлях(шлях, текст_коду.шлях::адреса);
    якщо р < 0 {
      // потім: почистити
      вихід_помилки::вміст = створити_помилку(компілятор, "Не вдалось клонувати шлях.", пусто);
      вернути ні;
    }

    текст_коду.значення = значення_кд;
    текст_коду.оригінальне_значення = оригінальне_значення;

    вернути скомпілювати_текст_коду(компілятор, середовище, текст_коду, вихід_помилки);
  }

  стала СИМВОЛ_ПЛЮС = КД::символи::Хрестик;

  дія спробувати_розібрати_ціле_число(компілятор: адреса<Компілятор>, значення: кд, вихід: адреса<ц64>, вихід_помилки: адреса<кд>, вихід_позиції_помилки: адреса<природне>) -> логічне {
    змінна вихідне: ціле = 0;
    змінна знак: ц8 = 1;
    змінна п: природне = 0;

    якщо п < значення.розмір {
      якщо значення.дані[п] == СИМВОЛ_МІНУС {
        знак = ц8(0) - 1;
        п += 1;
      } інакше якщо значення.дані[п] == СИМВОЛ_ПЛЮС {
        п += 1;
      }
    }

    поки п < значення.розмір {
      змінна с = значення.дані[п];

      якщо с == СИМВОЛ_0 {
        с = 0;
      } інакше якщо с == СИМВОЛ_1 {
        с = 1;
      } інакше якщо с == СИМВОЛ_2 {
        с = 2;
      } інакше якщо с == СИМВОЛ_3 {
        с = 3;
      } інакше якщо с == СИМВОЛ_4 {
        с = 4;
      } інакше якщо с == СИМВОЛ_5 {
        с = 5;
      } інакше якщо с == СИМВОЛ_6 {
        с = 6;
      } інакше якщо с == СИМВОЛ_7 {
        с = 7;
      } інакше якщо с == СИМВОЛ_8 {
        с = 8;
      } інакше якщо с == СИМВОЛ_9 {
        с = 9;
      } інакше {
        стрибнути успіх;
      }

      вихідне = вихідне * 10 + ціле(с);

      п += 1;
    }

  успіх:
    вихід::вміст = вихідне * ціле(знак);
    вернути так;
  }

  дія спробувати_розібрати_дробове_число(компілятор: адреса<Компілятор>, значення: кд, вихід: адреса<р64>, вихід_помилки: адреса<кд>, вихід_позиції_помилки: адреса<природне>) -> логічне {
    змінна число: р64 = 0.0;
    змінна п: природне = 0;
    змінна с: п8 = значення.дані[0];

    поки п < значення.розмір {
      якщо с == СИМВОЛ_0 { // 0
        число = число * 10;
      } інакше якщо с == СИМВОЛ_1 { // 1
        число = число * 10 + 1;
      } інакше якщо с == СИМВОЛ_2 { // 2
        число = число * 10 + 2;
      } інакше якщо с == СИМВОЛ_3 { // 3
        число = число * 10 + 3;
      } інакше якщо с == СИМВОЛ_4 { // 4
        число = число * 10 + 4;
      } інакше якщо с == СИМВОЛ_5 { // 5
        число = число * 10 + 5;
      } інакше якщо с == СИМВОЛ_6 { // 6
        число = число * 10 + 6;
      } інакше якщо с == СИМВОЛ_7 { // 7
        число = число * 10 + 7;
      } інакше якщо с == СИМВОЛ_8 { // 8
        число = число * 10 + 8;
      } інакше якщо с == СИМВОЛ_9 { // 9
        число = число * 10 + 9;
      } інакше якщо с == СИМВОЛ_КРАПКА { // .
        п += 1;
        с = значення.дані[п];

        змінна дробна_частина: р64 = 0.0;
        змінна дільник: р64 = 1.0;

        поки п < значення.розмір {
          якщо с == СИМВОЛ_0 { // 0
            дробна_частина = дробна_частина * 10;
          } інакше якщо с == СИМВОЛ_1 { // 1
            дробна_частина = дробна_частина * 10 + 1;
          } інакше якщо с == СИМВОЛ_2 { // 2
            дробна_частина = дробна_частина * 10 + 2;
          } інакше якщо с == СИМВОЛ_3 { // 3
            дробна_частина = дробна_частина * 10 + 3;
          } інакше якщо с == СИМВОЛ_4 { // 4
            дробна_частина = дробна_частина * 10 + 4;
          } інакше якщо с == СИМВОЛ_5 { // 5
            дробна_частина = дробна_частина * 10 + 5;
          } інакше якщо с == СИМВОЛ_6 { // 6
            дробна_частина = дробна_частина * 10 + 6;
          } інакше якщо с == СИМВОЛ_7 { // 7
            дробна_частина = дробна_частина * 10 + 7;
          } інакше якщо с == СИМВОЛ_8 { // 8
            дробна_частина = дробна_частина * 10 + 8;
          } інакше якщо с == СИМВОЛ_9 { // 9
            дробна_частина = дробна_частина * 10 + 9;
          } інакше {
            вихід_позиції_помилки::вміст = п;
            вернути ні;
          }

          п += 1;
          с = значення.дані[п];
          дільник = дільник * 10;
        }

        число = число + дробна_частина / дільник;

        вихід::вміст = число;
        вернути так;
      } інакше {
        вихід_позиції_помилки::вміст = п;
        вернути ні;
      }

      п += 1;
      с = значення.дані[п];
    }

    вихід::вміст = число;
    вернути так;
  }

  дія перетворити_природне_на_кд(компілятор: адреса<Компілятор>, значення: природне, вихід: адреса<кд>) -> логічне {
    змінна довжина: природне = 0;
    змінна тимч: природне = значення;

    довжина += 1;
    тимч /= 10;

    поки тимч != 0 {
      довжина += 1;
      тимч /= 10;
    }

    змінна значення_кд = кд { довжина, біб::виділити<п8>(довжина) };

    змінна п = ціле(довжина - 1);
    поки п >= 0 {
      значення_кд.дані[п] = п8(СИМВОЛ_0) + п8(значення % 10);
      значення /= 10;
      п -= 1;
    }

    вихід::вміст = значення_кд;

    вернути так;
  }

  дія отримати_повну_назву_в_контексті(компілятор: адреса<Компілятор>, контекст: Контекст, назва: кд) -> кд {
    змінна накопичувач_кд = зробити_накопичувач_кд(компілятор);

    покласти_в_накопичувач_кд(компілятор, накопичувач_кд::адреса, назва);

    змінна поточне_середовище = контекст.середовище;

    поки поточне_середовище != пусто {
      якщо поточне_середовище.обʼєкт_секції != пусто {
        якщо поточне_середовище.обʼєкт_секції.назва.значення.розмір > 0 {
          покласти_спереду_накопичувача_кд(компілятор, накопичувач_кд::адреса, "::");
          покласти_спереду_накопичувача_кд(компілятор, накопичувач_кд::адреса, поточне_середовище.обʼєкт_секції.назва.значення);
        }
      } інакше якщо поточне_середовище.обʼєкт_дії != пусто {
        якщо поточне_середовище.обʼєкт_дії.назва.значення.розмір > 0 {
          покласти_спереду_накопичувача_кд(компілятор, накопичувач_кд::адреса, "::");
          покласти_спереду_накопичувача_кд(компілятор, накопичувач_кд::адреса, поточне_середовище.обʼєкт_дії.назва.значення);
        }
      }

      поточне_середовище = поточне_середовище.зовнішнє;
    }

    вернути накопичувач_кд.значення;
  }
