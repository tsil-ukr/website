простір КЦ {
  дія обʼєкт_шаблону_виконати_шаблон(компілятор: адреса<Компілятор>, контекст: Контекст, місцезнаходження: розбирач::Місцезнаходження, обʼєкт_шаблону: адреса<ОбʼєктШаблону>, кількість_аргументів: природне, аргументи: адреса<Аргумент>) -> РезультатОбʼєкта {
    якщо кількість_аргументів != обʼєкт_шаблону.параметри.розмір {
      вернути результат_обʼєкта_помилка(компілятор, контекст.розібраний_текст, місцезнаходження,"Невірна кількість аргументів шаблону");
    }

    змінна контекст_виконання_шаблону = біб::виділити<КонтекстКомпіляціїВиконанняШаблону>(1);
    контекст_виконання_шаблону.обʼєкт_шаблону = обʼєкт_шаблону;
    контекст_виконання_шаблону.готові_аргументи_шаблону = ГотовіАргументиШаблону { кількість_аргументів, біб::виділити<ГотовийАргументШаблону>(кількість_аргументів) };
    змінна середовище_шаблону = створити_середовище(компілятор, обʼєкт_шаблону.контекст_компіляції.середовище);
    змінна п: природне = 0;
    поки п < кількість_аргументів {
      змінна аргумент = аргументи[п];
      змінна результат_компіляції_значення_аргумента = отримати_як_тип(компілятор, контекст, аргумент.місцезнаходження, аргумент.обʼєкт, ні);
      якщо результат_компіляції_значення_аргумента.помилка != пусто {
        знищити_готові_аргументи_шаблону(компілятор, контекст_виконання_шаблону.готові_аргументи_шаблону);
        біб::звільнити(контекст_виконання_шаблону);
        вернути РезультатОбʼєкта { ..., помилка = результат_компіляції_значення_аргумента.помилка };
      }
      контекст_виконання_шаблону.готові_аргументи_шаблону.дані[п] = ГотовийАргументШаблону {
        ВидАргументаГотовихШаблонуОбʼєктТипу,
        результат_компіляції_значення_аргумента.обʼєкт_типу
      };
      змінити_в_середовищі(
        середовище_шаблону,
        обʼєкт_шаблону.параметри.дані[п].назва,
        Обʼєкт { ВидОбʼєктаТип, дані = результат_компіляції_значення_аргумента.обʼєкт_типу }
      );
      п += 1;
    }
    змінна обʼєкт_з_реєстру_готових: Обʼєкт;

    якщо біб::знайти_в_реєстрі<ГотовіАргументиШаблону, Обʼєкт>(обʼєкт_шаблону.готові, контекст_виконання_шаблону.готові_аргументи_шаблону, обʼєкт_з_реєстру_готових::адреса) {
      знищити_готові_аргументи_шаблону(компілятор, контекст_виконання_шаблону.готові_аргументи_шаблону);
      біб::звільнити(контекст_виконання_шаблону);
      вернути РезультатОбʼєкта { обʼєкт_з_реєстру_готових, помилка = пусто };
    }

    змінна результат = скомпілювати_значення(
      компілятор,
      Контекст {
        розібраний_текст = обʼєкт_шаблону.контекст_компіляції.розібраний_текст,
        середовище = середовище_шаблону,
        кк_дії = обʼєкт_шаблону.контекст_компіляції.кк_дії
      },
      контекст_виконання_шаблону,
      обʼєкт_шаблону.сполука,
      пусто
    );

    знищити_готові_аргументи_шаблону(компілятор, контекст_виконання_шаблону.готові_аргументи_шаблону);
    біб::звільнити(контекст_виконання_шаблону);

    вернути результат;
  }
}
