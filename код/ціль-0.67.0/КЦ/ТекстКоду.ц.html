взяти визначення ./КЦ;

простір КЦ {
  дія створити_текст_коду(компілятор: адреса<Компілятор>) -> адреса<ТекстКоду> {
    змінна текст_коду = біб::виділити<ТекстКоду>(1);

    текст_коду.наступний = пусто;
    текст_коду.компілятор = компілятор;
    текст_коду.шлях = біб::Шлях { 0, пусто, 0 };
    текст_коду.значення = кд { 0, пусто };
    текст_коду.оригінальне_значення = пусто;

    якщо компілятор.перший_текст_коду == пусто {
      компілятор.перший_текст_коду = текст_коду;
    }
    якщо компілятор.останній_текст_коду == пусто {
      компілятор.останній_текст_коду = текст_коду;
    } інакше {
      компілятор.останній_текст_коду.наступний = текст_коду;
      компілятор.останній_текст_коду = текст_коду;
    }

    вернути текст_коду;
  }

  дія знищити_текст_коду(текст_коду: адреса<ТекстКоду>) {
    змінна компілятор = текст_коду.компілятор;

    біб::знищити_шлях(текст_коду.шлях);

    біб::звільнити(текст_коду.значення.дані);

    біб::звільнити(текст_коду);
  }
}
