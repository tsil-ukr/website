взяти визначення ./КЦ;

простір КЦ {
  внутрішня дія перед_заміною_в_реєстрі(реєстр: адреса<біб::Реєстр<біб::Шлях, адреса<РозібранийТекст>>>, ак: біб::Шлях, аз: адреса<РозібранийТекст>, бк: адреса<біб::Шлях>, бз: адреса<адреса<РозібранийТекст>>) {
    змінна компілятор = реєстр.аргумент як адреса<Компілятор>;

    біб::знищити_шлях(ак);
  }

  внутрішня дія перед_видаленням_з_реєстру(реєстр: адреса<біб::Реєстр<біб::Шлях, адреса<РозібранийТекст>>>, ключ: біб::Шлях, значення: адреса<РозібранийТекст>) {
    змінна компілятор = реєстр.аргумент як адреса<Компілятор>;

    біб::знищити_шлях(ключ);
  }

  дія створити_взяті_файли(компілятор: адреса<Компілятор>) -> адреса<ВзятіФайли> {
    змінна взяті_файли = біб::виділити<ВзятіФайли>(1);

    взяті_файли.компілятор = компілятор;
    взяті_файли.реєстр = біб::виділити_реєстр<біб::Шлях, адреса<РозібранийТекст>>(
      0,
      компілятор,
      змішати_шлях_для_реєстру<адреса<РозібранийТекст>>,
      перевірити_шляхи_в_реєстрі<адреса<РозібранийТекст>>,
      перед_заміною_в_реєстрі,
      перед_видаленням_з_реєстру
    );

    вернути взяті_файли;
  }

  дія знищити_взяті_файли(взяті_файли: адреса<ВзятіФайли>) {
    змінна компілятор = взяті_файли.компілятор;

    біб::звільнити_реєстр<біб::Шлях, адреса<РозібранийТекст>>(взяті_файли.реєстр);

    біб::звільнити(взяті_файли);
  }

  дія покласти_взятий_файл(взяті_файли: адреса<ВзятіФайли>, шлях: біб::Шлях, розібраний_текст: адреса<РозібранийТекст>) {
    змінна компілятор = взяті_файли.компілятор;

    біб::клонувати_шлях(шлях, шлях::адреса);

    біб::змінити_в_реєстрі<біб::Шлях, адреса<РозібранийТекст>>(
      взяті_файли.реєстр,
      шлях,
      розібраний_текст
    );
  }

  дія знайти_взятий_файл(взяті_файли: адреса<ВзятіФайли>, шлях: біб::Шлях, вихід: адреса<адреса<РозібранийТекст>>) -> логічне {
    вернути біб::знайти_в_реєстрі<біб::Шлях, адреса<РозібранийТекст>>(
      взяті_файли.реєстр,
      шлях,
      вихід
    );
  }
}
