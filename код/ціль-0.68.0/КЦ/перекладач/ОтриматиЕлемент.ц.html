взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_отримати_елемент(компілятор: адреса<Компілятор>, контекст: Контекст, дані_отримати_елемент: розбирач::ДаніСполукиОтриматиЕлемент, місцезнаходження: розбирач::Місцезнаходження, очікуваний_тип: адреса<ОбʼєктТипу>) -> РезультатОбʼєкта {
    змінна результат_ліво = скомпілювати_значення(компілятор, контекст, пусто, дані_отримати_елемент.ліво, пусто);
    якщо результат_ліво.помилка != пусто {
      вернути РезультатОбʼєкта { помилка = результат_ліво.помилка, ... };
    }

    змінна результат_ключа = скомпілювати_значення(компілятор, контекст, пусто, дані_отримати_елемент.ключ, пусто);
    якщо результат_ключа.помилка != пусто {
      вернути РезультатОбʼєкта { помилка = результат_ключа.помилка, ... };
    }

    змінна результат = обʼєкт_отримати_за_позицією(
      компілятор,
      контекст,
      місцезнаходження,
      результат_ліво.обʼєкт,
      Аргумент { результат_ключа.обʼєкт, дані_отримати_елемент.ключ.місцезнаходження },
      очікуваний_тип
    );

    вернути результат;
  }
}
