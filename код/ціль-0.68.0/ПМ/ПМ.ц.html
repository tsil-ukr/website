простір ПМ {
  дія створити_сховище_назв() -> адреса<СховищеНазв> {
    змінна сн = біб::виділити<СховищеНазв>(1);

    сн.розмір = 0;
    сн.дані = пусто;
    сн.вмісткість = 0;

    вернути сн;
  }

  дія знищити_сховище_назв(сховище: адреса<СховищеНазв>) {
    змінна п: природне = 0;
    поки п < сховище.розмір {
      біб::звільнити(сховище.дані[п].значення.дані);
      біб::звільнити(сховище.дані[п]);

      п += 1;
    }

    біб::звільнити(сховище.дані);
    біб::звільнити(сховище);
  }

  дія отримати_назву(сховище: адреса<СховищеНазв>, значення: кд) -> адреса<Назва> {
    змінна знайдена_назва: адреса<Назва> = пусто;

    змінна п: природне = 0;
    поки п < сховище.розмір {
      якщо КД::перевірити_чи_кд_рівні(сховище.дані[п].значення, значення) {
        вернути сховище.дані[п];
      }

      п += 1;
    }

    змінна нова_назва = біб::виділити<Назва>(1);

    нова_назва.значення = кд {
      значення.розмір,
      біб::виділити<п8>(значення.розмір)
    };

    п = 0;
    поки п < значення.розмір {
      нова_назва.значення.дані[п] = значення.дані[п];

      п += 1;
    }

    якщо сховище.розмір == сховище.вмісткість {
      сховище.вмісткість += 1;
      сховище.дані = біб::перевиділити<адреса<Назва>>(сховище.дані, сховище.вмісткість);
    }

    сховище.дані[сховище.розмір] = нова_назва;
    сховище.розмір += 1;

    вернути нова_назва;
  }

  // потім: це працює неправильно
  дія отримати_назву_без_копіювання(сховище: адреса<СховищеНазв>, значення: кд) -> адреса<Назва> {
    змінна знайдена_назва: адреса<Назва> = пусто;

    змінна п: природне = 0;
    поки п < сховище.розмір {
      якщо КД::перевірити_чи_кд_рівні(сховище.дані[п].значення, значення) {
        // біб::звільнити(значення.дані);
                                                        //        треба тут виправити в майбутньому

        вернути сховище.дані[п];
      }

      п += 1;
    }

    змінна нова_назва = біб::виділити<Назва>(1);

    нова_назва.значення = значення;

    якщо сховище.розмір == сховище.вмісткість {
      сховище.вмісткість += 1;
      сховище.дані = біб::перевиділити<адреса<Назва>>(сховище.дані, сховище.вмісткість);
    }

    сховище.дані[сховище.розмір] = нова_назва;
    сховище.розмір += 1;

    вернути нова_назва;
  }
}

простір ПМ {
  внутрішня дія покласти_крок(інструкція: адреса<Інструкція>, крок: адреса<Крок>) {
    змінна пм_значення = інструкція як адреса<Значення>;
    змінна дані_інструкції = пм_значення.дані як ДаніЗначенняІнструкція;

    якщо дані_інструкції.кроки.розмір == дані_інструкції.кроки.вмісткість {
      дані_інструкції.кроки.вмісткість += 1;
      дані_інструкції.кроки.дані = біб::перевиділити<адреса<Крок>>(дані_інструкції.кроки.дані, дані_інструкції.кроки.вмісткість);
    }

    дані_інструкції.кроки.дані[дані_інструкції.кроки.розмір] = крок;
    дані_інструкції.кроки.розмір += 1;

    пм_значення.дані = дані_інструкції;
  }

  дія покласти_параметр(модуль: адреса<Модуль>, параметри: адреса<Параметри>, параметр: Параметр) {
    якщо параметри.розмір == параметри.вмісткість {
      параметри.вмісткість += 1;
      параметри.дані = біб::перевиділити<Параметр>(параметри.дані, параметри.вмісткість);
    }

    параметри.дані[параметри.розмір] = параметр;
    параметри.розмір += 1;
  }

  внутрішня дія виділити_тип(модуль: адреса<Модуль>, вид: природне) -> адреса<Тип> {
    змінна тип = біб::виділити<Тип>(1);

    тип.модуль = модуль;
    тип.вид = вид;
    тип.дані_компіляції = пусто;

    вернути тип;
  }

  внутрішня дія виділити_значення(модуль: адреса<Модуль>, вид: природне, тип: адреса<Тип>) -> адреса<Значення> {
    змінна значення = біб::виділити<Значення>(1);

    значення.модуль = модуль;
    значення.вид = вид;
    значення.тип = тип;
    значення.дані_компіляції = пусто;

    вернути значення;
  }

  внутрішня дія змішати_назви_реєстру<Т>(реєстр: адреса<біб::Реєстр<адреса<Назва>, Т>>, назва: адреса<Назва>) -> природне {
    вернути назва.значення.розмір;
  }

  внутрішня дія перевірити_назви_реєстру<Т>(реєстр: адреса<біб::Реєстр<адреса<Назва>, Т>>, а: адреса<Назва>, б: адреса<Назва>) -> логічне {
    вернути а == б;
  }

  внутрішня дія виділити_реєстр_по_назвах<Т>() -> адреса<біб::Реєстр<адреса<Назва>, Т>> {
    вернути біб::виділити_реєстр<адреса<Назва>, Т>(
      0,
      пусто,
      змішати_назви_реєстру<Т>,
      перевірити_назви_реєстру<Т>,
      пусто,
      пусто
    );
  }

  дія створити_модуль(назва: адреса<Назва>, платформа: природне) -> адреса<Модуль> {
    змінна модуль = біб::виділити<Модуль>(1);

    модуль.платформа = платформа;

    модуль.типи = виділити_реєстр_по_назвах<адреса<Тип>>();
    модуль.глобальні = виділити_реєстр_по_назвах<адреса<Значення>>();

    модуль.тип_ніщо = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_адреса = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_логічне = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_п8 = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_п16 = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_п32 = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_п64 = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_ц8 = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_ц16 = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_ц32 = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_ц64 = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_р32 = виділити_тип(модуль, ВидТипуНеподільний);
    модуль.тип_р64 = виділити_тип(модуль, ВидТипуНеподільний);

    модуль.значення_пусто = виділити_значення(модуль, ВидЗначенняЧисло, модуль.тип_адреса);
    модуль.значення_пусто.дані = п64(0);

    модуль.значення_так = виділити_значення(модуль, ВидЗначенняЧисло, модуль.тип_логічне);
    модуль.значення_так.дані = так;

    модуль.значення_ні = виділити_значення(модуль, ВидЗначенняЧисло, модуль.тип_логічне);
    модуль.значення_ні.дані = ні;

    вернути модуль;
  }

  дія створити_інструкцію(модуль: адреса<Модуль>, видимість: природне, назва: адреса<Назва>, параметри: Параметри, тип_результату: адреса<Тип>) -> адреса<Інструкція> {
    змінна тип_інструкції = створити_тип_інструкції(модуль, параметри, тип_результату);
    змінна значення_інструкції = виділити_значення(модуль, ВидЗначенняІнструкція, тип_інструкції);

    змінна значення_аргументів = біб::виділити<адреса<Значення>>(параметри.розмір);
    змінна п: природне = 0;
    поки п < параметри.розмір {
      значення_аргументів[п] = виділити_значення(модуль, ВидЗначенняАргумент, параметри.дані[п].тип);
      значення_аргументів[п].дані = ДаніЗначенняАргумент { п, значення_інструкції };

      п += 1;
    }

    значення_інструкції.дані = ДаніЗначенняІнструкція {
      модуль = модуль,
      видимість = видимість,
      назва = назва,
      тип = тип_інструкції,
      кроки = Кроки { 0, пусто, 0 },
      значення_аргументів = ЗначенняАргументів { параметри.розмір, значення_аргументів, параметри.розмір },
    };

    біб::змінити_в_реєстрі<адреса<Назва>, адреса<Значення>>(модуль.глобальні, назва, значення_інструкції);

    вернути значення_інструкції як адреса<Інструкція>;
  }

  дія отримати_тип_інструкції(інструкція: адреса<Інструкція>) -> адреса<Тип> {
    змінна значення_інструкції = інструкція як адреса<Значення>;
    змінна дані_інструкції = значення_інструкції.дані як ДаніЗначенняІнструкція;

    вернути дані_інструкції.тип;
  }

  дія створити_тип_інструкції(модуль: адреса<Модуль>, параметри: Параметри, тип_результату: адреса<Тип>) -> адреса<Тип> {
    змінна тип = виділити_тип(модуль, ВидТипуІнструкція);

    тип.дані = ДаніТипуІнструкція { параметри, тип_результату };

    вернути тип;
  }

  дія отримати_значення_аргумента_інструкції(інструкція: адреса<Інструкція>, позиція: природне) -> адреса<Значення> {
    змінна значення_інструкції = інструкція як адреса<Значення>;
    змінна дані_інструкції = значення_інструкції.дані як ДаніЗначенняІнструкція;

    вернути дані_інструкції.значення_аргументів.дані[позиція];
  }

  дія створити_крок(інструкція: адреса<Інструкція>, назва: адреса<Назва>) -> адреса<Крок> {
    змінна значення_інструкції = інструкція як адреса<Значення>;
    змінна дані_інструкції = значення_інструкції.дані як ДаніЗначенняІнструкція;

    змінна крок = біб::виділити<Крок>(1);

    крок.інструкція = інструкція;
    крок.назва = назва;
    крок.вказівки = Вказівки { 0, пусто, 0 };
    крок.дані_компіляції = пусто;

    покласти_крок(інструкція, крок);

    вернути крок;
  }

  дія перевірити_чи_крок_закінчується_стрибком(крок: адреса<Крок>) -> логічне {
    якщо крок.вказівки.розмір > 0 {
      якщо крок.вказівки.дані[крок.вказівки.розмір - 1].вид == ВидВказівкиСтрибнути {
        вернути так;
      }
    }

    вернути ні;
  }

  дія створити_структуру(модуль: адреса<Модуль>, назва: адреса<Назва>, параметри: Параметри, пакована: логічне) -> адреса<Тип> {
    змінна тип = виділити_тип(модуль, ВидТипуСтруктура);

    тип.дані = ДаніТипуСтруктура { назва, параметри, пакована };

    біб::змінити_в_реєстрі<адреса<Назва>, адреса<Тип>>(модуль.типи, назва, тип);

    вернути тип;
  }

  дія заповнити_параметри_структури(тип: адреса<Тип>, параметри: Параметри) {
    якщо тип.вид != ВидТипуСтруктура {
      вернути;
    }

    змінна дані = тип.дані як ДаніТипуСтруктура;
    дані.параметри = параметри;

    тип.дані = дані;
  }

  внутрішня дія отримати_модуль_з_інструкції(інструкція: адреса<Інструкція>) -> адреса<Модуль> {
    змінна значення_інструкції = інструкція як адреса<Значення>;
    змінна дані_інструкції = значення_інструкції.дані як ДаніЗначенняІнструкція;

    вернути дані_інструкції.модуль;
  }

  внутрішня дія покласти_вказівку(крок: адреса<Крок>, вказівка: адреса<Значення>) {
    якщо крок.вказівки.розмір == крок.вказівки.вмісткість {
      крок.вказівки.вмісткість += 1;
      крок.вказівки.дані = біб::перевиділити<адреса<Значення>>(крок.вказівки.дані, крок.вказівки.вмісткість);
    }

    крок.вказівки.дані[крок.вказівки.розмір] = вказівка;
    крок.вказівки.розмір += 1;
  }

  дія виконати_дію(крок: адреса<Крок>, тип: адреса<Тип>, значення: адреса<Значення>, кількість_аргументів: природне, аргументи: адреса<адреса<Значення>>) -> адреса<Значення> {
    змінна модуль = отримати_модуль_з_інструкції(крок.інструкція);
    змінна дані_типу_інструкція = тип.дані як ДаніТипуІнструкція;

    змінна копія_аргументів = біб::виділити<адреса<Значення>>(кількість_аргументів);
    змінна п: природне = 0;
    поки п < кількість_аргументів {
      копія_аргументів[п] = аргументи[п];

      п += 1;
    }

    змінна значення_вказівки = виділити_значення(модуль, ВидЗначенняВказівка, дані_типу_інструкція.тип_результату);

    значення_вказівки.дані = ДаніЗначенняВказівка {
      крок = крок,
      вид = ВидВказівкиВиконати,
      дані = ДаніВказівкиВиконати { тип, значення, кількість_аргументів, копія_аргументів }
    };

    покласти_вказівку(крок, значення_вказівки);

    вернути значення_вказівки;
  }

  внутрішня дія _отримати_розмір_типу_для_виділення(модуль: адреса<Модуль>, тип: адреса<Тип>, вихід_вирівнювання: адреса<природне>) -> природне {
    якщо тип == пусто {
      біб::надрукувати("[ПОМИЛКА КОМПІЛЯТОРА] тип == пусто в _отримати_розмір_типу_для_виділення");
      вернути 0;
    }

    якщо тип == модуль.тип_логічне {
      вихід_вирівнювання::вміст = 1;
      вернути 1;
    } інакше якщо тип == модуль.тип_п8 {
      вихід_вирівнювання::вміст = 1;
      вернути 1;
    } інакше якщо тип == модуль.тип_п16 {
      вихід_вирівнювання::вміст = 2;
      вернути 2;
    } інакше якщо тип == модуль.тип_п32 {
      вихід_вирівнювання::вміст = 4;
      вернути 4;
    } інакше якщо тип == модуль.тип_п64 {
      вихід_вирівнювання::вміст = 8;
      вернути 8;
    } інакше якщо тип == модуль.тип_ц8 {
      вихід_вирівнювання::вміст = 1;
      вернути 1;
    } інакше якщо тип == модуль.тип_ц16 {
      вихід_вирівнювання::вміст = 2;
      вернути 2;
    } інакше якщо тип == модуль.тип_ц32 {
      вихід_вирівнювання::вміст = 4;
      вернути 4;
    } інакше якщо тип == модуль.тип_ц64 {
      вихід_вирівнювання::вміст = 8;
      вернути 8;
    } інакше якщо тип == модуль.тип_р32 {
      вихід_вирівнювання::вміст = 4;
      вернути 4;
    } інакше якщо тип == модуль.тип_р64 {
      вихід_вирівнювання::вміст = 8;
      вернути 8;
    } інакше якщо тип == модуль.тип_адреса {
      вихід_вирівнювання::вміст = 8;
      вернути 8;
    } інакше якщо тип.вид == ВидТипуСтруктура {
      змінна дані_структури = тип.дані як ДаніТипуСтруктура;

      змінна зсув: природне = 0;
      змінна макс_вирівнювання: природне = 1;

      змінна п: природне = 0;

      поки п < дані_структури.параметри.розмір {
        змінна параметр = дані_структури.параметри.дані[п];
        змінна тип = параметр.тип;

        змінна вирівнювання: природне = 0;
        змінна розмір_типу = _отримати_розмір_типу_для_виділення(модуль, тип, вирівнювання::адреса);

        якщо не дані_структури.пакована і вирівнювання > макс_вирівнювання {
          макс_вирівнювання = вирівнювання;
        }

        змінна викор_вирівн: природне = 1;
        якщо не дані_структури.пакована {
          викор_вирівн = вирівнювання;
        }

        змінна доповнення = (викор_вирівн - (зсув % викор_вирівн)) % викор_вирівн;

        зсув += (доповнення + розмір_типу);

        п += 1;
      }

      змінна кінцеве_вирівнювання: природне = 1;
      якщо не дані_структури.пакована {
        кінцеве_вирівнювання = макс_вирівнювання;
      }

      змінна кінцеве_доповнення = (кінцеве_вирівнювання - (зсув % кінцеве_вирівнювання)) % кінцеве_вирівнювання;
      зсув += кінцеве_доповнення;

      вихід_вирівнювання::вміст = кінцеве_вирівнювання;
      вернути зсув;
    }

    біб::надрукувати("[ПОМИЛКА КОМПІЛЯТОРА] невідомий тип в _отримати_розмір_типу_для_виділення");
    вернути 0;
  }

    змінна вирівнювання: природне = 0;
    вернути _отримати_розмір_типу_для_виділення(модуль, тип, вирівнювання::адреса);
  }

  дія вказівка_отримати_адресу_елемента(крок: адреса<Крок>, тип: адреса<Тип>, значення: адреса<Значення>, кількість_позицій: природне, позиції: адреса<адреса<Значення>>) -> адреса<Значення> {

    змінна пм_значення = біб::виділити<Значення>(1);

    змінна копія_позиції = біб::виділити<адреса<Значення>>(кількість_позицій);

    змінна п: природне = 0;
    поки п < кількість_позицій {
      копія_позиції[п] = позиції[п];

      п += 1;
    }

    пм_значення.модуль = модуль;
    пм_значення.вид = ВидЗначенняВказівка;
    пм_значення.тип = отримати_модуль_з_інструкції(крок.інструкція).тип_адреса;

      крок = крок,
      вид = ВидВказівкиОтриматиАдресуЕлемента,
      дані = ДаніВказівкиОтриматиАдресуЕлемента { тип, значення, кількість_позицій, копія_позиції }
    };
    пм_значення.дані_компіляції = пусто;

    покласти_вказівку(крок, пм_значення);

    вернути пм_значення;
