  дія виділити_обʼєкт_сирого_значення(компілятор: *Компілятор) -> *ОбʼєктСирогоЗначення {
    змінна обʼєкт_сирого_значення = біб::виділити<ОбʼєктСирогоЗначення>(1);

    обʼєкт_сирого_значення.наступний = пусто;
    обʼєкт_сирого_значення.компілятор = компілятор;
    обʼєкт_сирого_значення.вид = ВидСирогоОбʼєктаЧисло;
    обʼєкт_сирого_значення.дані = кд { пусто, 0 };

    якщо компілятор.перший_обʼєкт_сирого_значення == пусто {
      компілятор.перший_обʼєкт_сирого_значення = обʼєкт_сирого_значення;
    }
    якщо компілятор.останній_обʼєкт_сирого_значення == пусто {
      компілятор.останній_обʼєкт_сирого_значення = обʼєкт_сирого_значення;
    } інакше {
      компілятор.останній_обʼєкт_сирого_значення.наступний = обʼєкт_сирого_значення;
      компілятор.останній_обʼєкт_сирого_значення = обʼєкт_сирого_значення;
    }

    вернути обʼєкт_сирого_значення;
  }

    біб::звільнити(обʼєкт_сирого_значення.дані.дані);

    біб::звільнити(обʼєкт_сирого_значення);
  }
}

  дія обʼєкт_сирого_значення_як(компілятор: *Компілятор, контекст: Контекст, місцезнаходження: розбирач::Місцезнаходження, обʼєкт_сирого_значення: *ОбʼєктСирогоЗначення, аргумент: Аргумент) -> РезультатОбʼєкта {
    якщо аргумент.обʼєкт.вид == ВидОбʼєктаТип {
      змінна тип = аргумент.обʼєкт.дані як *ОбʼєктТипу;

      якщо обʼєкт_сирого_значення.вид == ВидСирогоОбʼєктаПусто {
        змінна результат = створити_пустий_обʼєкт(компілятор, контекст, місцезнаходження, тип);

        якщо результат.помилка != пусто {
          вернути РезультатОбʼєкта { помилка = результат.помилка, ... };
        }

        вернути РезультатОбʼєкта {
          Обʼєкт {
            ВидОбʼєктаЗначення,
            результат.обʼєкт_значення
          },
          помилка = пусто
        };
      }

      якщо обʼєкт_сирого_значення.вид == ВидСирогоОбʼєктаЧисло {
        змінна результат_розбору_числа = розібрати_число(
          компілятор,
          контекст,
          місцезнаходження,
          обʼєкт_сирого_значення.дані,
          тип
        );

        якщо результат_розбору_числа.помилка != пусто {
          вернути РезультатОбʼєкта { ..., помилка = результат_розбору_числа.помилка };
        }

        вернути РезультатОбʼєкта {
          Обʼєкт {
            ВидОбʼєктаЗначення,
            результат_розбору_числа.обʼєкт_значення
          },
          помилка = пусто
        };
      }
    }

    вернути результат_обʼєкта_помилка(компілятор, контекст.розібраний_текст, місцезнаходження, "Не вдалось скомпілювати");
  }
}
