взяти визначення ./ПМ;
взяти визначення ./ПМЛЛВМ;
взяти визначення ./КЦ;
взяти визначення ./команди;
взяти біб цілеспрямовано;

дія почати(кількість_аргументів: природне, аргументи: *ю8) -> ц32 {
  змінна статус_виходу: ц32 = 0;

  змінна кількість_аргументів_кд = кількість_аргументів - 1;
  змінна аргументи_кд = біб::виділити<кд>(кількість_аргументів_кд);
  змінна п: природне = 0;
  поки п < кількість_аргументів_кд {
    змінна позиція_помилки: природне = 0;

    якщо не біб::перекодувати_з_ю8(аргументи[п + 1], ні, так, аргументи_кд[п]::адреса, пусто, позиція_помилки::адреса) {
      біб::вивести("Не вдалось перекодувати аргумент \"");
      біб::вивести_ю8(аргументи[п + 1]);
      біб::надрукувати("\" у кд.");
      вернути 1;
    }

    п += 1;
  }

  змінна кількість_команд: природне = 8;

  змінна доступні_команди = цілеспрямовано::ДоступніКомандиДляРозбору<кд> { кількість_команд, біб::виділити<кд>(кількість_команд) };
  доступні_команди.дані[0] = "допомога";
  доступні_команди.дані[1] = "версія";
  доступні_команди.дані[2] = "скомпілювати";
  доступні_команди.дані[3] = "перетворити";
  доступні_команди.дані[4] = "ювкд";
  доступні_команди.дані[5] = "кдвю";
  доступні_команди.дані[6] = "платформи";
  доступні_команди.дані[7] = "зібрати";

  змінна розібрана_команда = цілеспрямовано::зробити_розібрану_команду<кд>();

  змінна успіх_розбору_команди = цілеспрямовано::розібрати_команду<кд>(
    цілеспрямовано::АргументиДляРозбору<кд> { кількість_аргументів_кд, аргументи_кд },
    доступні_команди,
    "--",
    "-",
    "=",
    розібрана_команда::адреса
  );

  якщо не успіх_розбору_команди {
    біб::надрукувати("Не вдалось розібрати команду!");
    вернути 1;
  }

  якщо біб::перевірити_чи_кд_рівні(розібрана_команда.команда, "") {
    статус_виходу = ціль_допомога(
      цілеспрямовано::Опції<кд> { 0, пусто },
      цілеспрямовано::Опції<кд> { 0, пусто }
    );

    стрибнути вихід;
  }

  якщо біб::перевірити_чи_кд_рівні(розібрана_команда.команда, "допомога") {
    статус_виходу = ціль_допомога(
      розібрана_команда.опції_програми,
      розібрана_команда.опції_команди
    );

    стрибнути вихід;
  }

  якщо біб::перевірити_чи_кд_рівні(розібрана_команда.команда, "версія") {
    статус_виходу = ціль_версія(
      розібрана_команда.опції_програми,
      розібрана_команда.опції_команди
    );

    стрибнути вихід;
  }

  якщо біб::перевірити_чи_кд_рівні(розібрана_команда.команда, "перетворити") або біб::перевірити_чи_кд_рівні(розібрана_команда.команда, "скомпілювати") {
    якщо розібрана_команда.входи.розмір < 1 {
      біб::надрукувати("Недостатньо вхідних!");
      статус_виходу = 1;
      стрибнути вихід;
    }

    якщо розібрана_команда.виходи.розмір == 0 {
      змінна п: природне = 0;
      поки п < розібрана_команда.входи.розмір {
        змінна вхід = розібрана_команда.входи.дані[п];

        статус_виходу = ціль_перетворити(
          розібрана_команда.опції_програми,
          розібрана_команда.опції_команди,
          вхід.значення,
          вхід.опції,
          кд { пусто, 0 },
          цілеспрямовано::Опції<кд> { 0, пусто },
          розібрана_команда.входи.розмір == 1
        );

        якщо статус_виходу != 0 {
          стрибнути вихід;
        }

        п += 1;
      }

      стрибнути вихід;
    } інакше {
      якщо розібрана_команда.входи.розмір > 1 {
        біб::надрукувати("Команда перетворити з вказаними виходами може мати лише один вхід!");
        статус_виходу = 1;
        стрибнути вихід;
      }

      змінна вхід = розібрана_команда.входи.дані[0];

      змінна п: природне = 0;
      поки п < розібрана_команда.виходи.розмір {
        змінна вихід = розібрана_команда.виходи.дані[п];

        статус_виходу = ціль_перетворити(
          розібрана_команда.опції_програми,
          розібрана_команда.опції_команди,
          вхід.значення,
          вхід.опції,
          вихід.значення,
          вихід.опції,
          розібрана_команда.виходи.розмір == 1
        );

        якщо статус_виходу != 0 {
          стрибнути вихід;
        }

        п += 1;
      }

      стрибнути вихід;
    }
  }

  якщо біб::перевірити_чи_кд_рівні(розібрана_команда.команда, "зібрати") {
    якщо розібрана_команда.входи.розмір < 1 {
      біб::надрукувати("Недостатньо вхідних!");
      статус_виходу = 1;
      стрибнути вихід;
    }

    якщо розібрана_команда.виходи.розмір == 0 {
      статус_виходу = ціль_зібрати(
        розібрана_команда.опції_програми,
        розібрана_команда.опції_команди,
        розібрана_команда.входи,
        кд { пусто, 0 },
        цілеспрямовано::Опції<кд> { 0, пусто },
      );

      стрибнути вихід;
    } інакше якщо розібрана_команда.виходи.розмір == 1 {
      статус_виходу = ціль_зібрати(
        розібрана_команда.опції_програми,
        розібрана_команда.опції_команди,
        розібрана_команда.входи,
        розібрана_команда.виходи.дані[0].значення,
        розібрана_команда.виходи.дані[0].опції,
      );

      стрибнути вихід;
    } інакше {
      біб::надрукувати("Команда зібрати може мати лише один вихід!");
      статус_виходу = 1;
      стрибнути вихід;
    }
  }

  якщо біб::перевірити_чи_кд_рівні(розібрана_команда.команда, "ювкд") {
    якщо розібрана_команда.входи.розмір < 1 {
      біб::надрукувати("Недостатньо вхідних!");
      статус_виходу = 1;
      стрибнути вихід;
    }

    якщо розібрана_команда.виходи.розмір == 0 {
      змінна п: природне = 0;
      поки п < розібрана_команда.входи.розмір {
        змінна вхід = розібрана_команда.входи.дані[п];

        статус_виходу = ціль_ювкд(
          розібрана_команда.опції_програми,
          розібрана_команда.опції_команди,
          вхід.значення,
          вхід.опції,
          кд { пусто, 0 },
          цілеспрямовано::Опції<кд> { 0, пусто },
          розібрана_команда.входи.розмір == 1
        );

        якщо статус_виходу != 0 {
          стрибнути вихід;
        }

        п += 1;
      }

      стрибнути вихід;
    } інакше {
      якщо розібрана_команда.входи.розмір > 1 {
        біб::надрукувати("Команда кдвю з вказаними виходами може мати лише один вхід!");
        статус_виходу = 1;
        стрибнути вихід;
      }

      змінна вхід = розібрана_команда.входи.дані[0];

      змінна п: природне = 0;
      поки п < розібрана_команда.виходи.розмір {
        змінна вихід = розібрана_команда.виходи.дані[п];

        статус_виходу = ціль_ювкд(
          розібрана_команда.опції_програми,
          розібрана_команда.опції_команди,
          вхід.значення,
          вхід.опції,
          вихід.значення,
          вихід.опції,
          розібрана_команда.виходи.розмір == 1
        );

        якщо статус_виходу != 0 {
          стрибнути вихід;
        }

        п += 1;
      }

      стрибнути вихід;
    }
  }

  якщо біб::перевірити_чи_кд_рівні(розібрана_команда.команда, "кдвю") {
    якщо розібрана_команда.входи.розмір < 1 {
      біб::надрукувати("Недостатньо вхідних!");
      статус_виходу = 1;
      стрибнути вихід;
    }

    якщо розібрана_команда.виходи.розмір == 0 {
      змінна п: природне = 0;
      поки п < розібрана_команда.входи.розмір {
        змінна вхід = розібрана_команда.входи.дані[п];

        статус_виходу = ціль_кдвю(
          розібрана_команда.опції_програми,
          розібрана_команда.опції_команди,
          вхід.значення,
          вхід.опції,
          кд { пусто, 0 },
          цілеспрямовано::Опції<кд> { 0, пусто },
          розібрана_команда.входи.розмір == 1
        );

        якщо статус_виходу != 0 {
          стрибнути вихід;
        }

        п += 1;
      }

      стрибнути вихід;
    } інакше {
      якщо розібрана_команда.входи.розмір > 1 {
        біб::надрукувати("Команда кдвю з вказаними виходами може мати лише один вхід!");
        статус_виходу = 1;
        стрибнути вихід;
      }

      змінна вхід = розібрана_команда.входи.дані[0];

      змінна п: природне = 0;
      поки п < розібрана_команда.виходи.розмір {
        змінна вихід = розібрана_команда.виходи.дані[п];

        статус_виходу = ціль_кдвю(
          розібрана_команда.опції_програми,
          розібрана_команда.опції_команди,
          вхід.значення,
          вхід.опції,
          вихід.значення,
          вихід.опції,
          розібрана_команда.виходи.розмір == 1
        );

        якщо статус_виходу != 0 {
          стрибнути вихід;
        }

        п += 1;
      }

      стрибнути вихід;
    }
  }

  якщо біб::перевірити_чи_кд_рівні(розібрана_команда.команда, "платформи") {
    статус_виходу = ціль_платформи(
      розібрана_команда.опції_програми,
      розібрана_команда.опції_команди
    );

    стрибнути вихід;
  }

  біб::надрукувати("Невідома команда!");
  статус_виходу = 1;

вихід:
  цілеспрямовано::знищити_розібрану_команду<кд>(розібрана_команда);
  біб::звільнити(доступні_команди.дані);
  вернути статус_виходу;
}
