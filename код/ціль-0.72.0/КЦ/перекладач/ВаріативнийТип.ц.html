взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_варіативний_тип(компілятор: *Компілятор, контекст: Контекст, дані_варіативного_типу: розбирач::ДаніСполукиВаріативнийТип, місцезнаходження: розбирач::Місцезнаходження) -> РезультатОбʼєктаТипу {
    змінна типи = біб::виділити<*ОбʼєктТипу>(дані_варіативного_типу.типи.розмір);

    змінна п: природне = 0;

    поки п < дані_варіативного_типу.типи.розмір {
      змінна результат_типу = скомпілювати_значення_як_тип(
        компілятор,
        контекст,
        дані_варіативного_типу.типи.дані[п],
        так
      );

      якщо результат_типу.помилка != пусто {
        біб::звільнити(типи);

        вернути результат_типу;
      }

      типи[п] = результат_типу.обʼєкт_типу;

      п += 1;
    }

    змінна обʼєкт_типу_варіації = створити_варіативний_тип(
      компілятор,
      дані_варіативного_типу.типи.розмір,
      типи
    );

    вернути РезультатОбʼєктаТипу {
      обʼєкт_типу_варіації,
      помилка = пусто
    };
  }
}
