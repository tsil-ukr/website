              компілятор,
              "Не вдалось отримати директорію шляху до файлу.",
              створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
            ),
            ...
          };

          ...

            помилка = створити_помилку(
              компілятор,
              "Шлях до біб не вказано.",
              створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
            ),
            ...

        }

    покласти_кд_в_накопичувач_шляху(компілятор, накопичувач_шляху_1::адреса, "/");

      біб::знищити_шлях(директорія);
    }

    поки п < дані_взяти.шлях.розмір {
      змінна частина_кд = кд { пусто, 0 };

      якщо не розбирач::отримати_значення_слова(
        контекст.розібраний_текст.текст_коду.значення,
        контекст.розібраний_текст.слова.дані[дані_взяти.шлях.дані[п].позиція_слова],
        частина_кд::адреса
      ) {
        вернути РезультатОбʼєкта {

          ...
        };

      покласти_кд_в_накопичувач_шляху(компілятор, накопичувач_шляху_3::адреса, частина_кд);
      покласти_кд_в_накопичувач_шляху(компілятор, накопичувач_шляху_4::адреса, частина_кд);

        покласти_кд_в_накопичувач_шляху(компілятор, накопичувач_шляху_2::адреса, "/");
        покласти_кд_в_накопичувач_шляху(компілятор, накопичувач_шляху_3::адреса, "/");
        покласти_кд_в_накопичувач_шляху(компілятор, накопичувач_шляху_4::адреса, "/");

        покласти_кд_в_накопичувач_шляху(компілятор, накопичувач_шляху_3::адреса, "/");
        покласти_кд_в_накопичувач_шляху(компілятор, накопичувач_шляху_3::адреса, частина_кд);

      біб::знищити_шлях(шлях_4);

              }

              якщо пп == дані_взяти.шлях.розмір - 1 {
                покласти_кд_в_накопичувач_шляху(компілятор, накопичувач_шляху_2::адреса, "/");
                покласти_кд_в_накопичувач_шляху(компілятор, накопичувач_шляху_2::адреса, частина_кд);
              }

            біб::шлях_з_ю8(ю8 { накопичувач_шляху_1.значення.дані, накопичувач_шляху_1.значення.розмір }, шлях_1::адреса);
            біб::звільнити(накопичувач_шляху_1.значення.дані);

            змінна шлях_2: біб::Шлях;
            біб::шлях_з_ю8(ю8 { накопичувач_шляху_2.значення.дані, накопичувач_шляху_2.значення.розмір }, шлях_2::адреса);

                вернути РезультатОбʼєкта { помилка = помилка, ... };

              біб::абсолютний_шлях(шлях_2, абсолютний_шлях_2::адреса);
