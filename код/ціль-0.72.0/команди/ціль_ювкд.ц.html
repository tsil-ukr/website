взяти біб цілеспрямовано;
взяти біб Памʼять;
взяти біб КД;
взяти визначення КЦ;
взяти визначення ./команди;

дія ціль_ювкд(опції_програми: цілеспрямовано::Опції<кд>,
                      опції_команди: цілеспрямовано::Опції<кд>,
                      вхід: кд,
                      опції_входу: цілеспрямовано::Опції<кд>,
                      вихід: кд,
                      опції_виходу: цілеспрямовано::Опції<кд>,
                      треба_знищити_залишки: логічне) -> ц32 {
  змінна р: ціле;
  змінна статус: ц32 = 0;

  змінна шлях_входу = біб::пустий_шлях();
  змінна шлях_виходу = біб::пустий_шлях();
  змінна шлях_виходу_місцезнаходжень = біб::пустий_шлях();
  змінна байти_входу = біб::Інформація { 0, пусто, 0 };
  змінна значення_виходу = кд { пусто, 0 };

  змінна вихід_місцезнаходжень = кд { пусто, 0 };
  змінна визначено_вихід_місцезнаходжень = ні;

  змінна п: природне = 0;

  поки п < опції_виходу.розмір {
    змінна опція = опції_виходу.дані[п];

    якщо біб::перевірити_чи_кд_рівні(опція.ключ, "місцезнаходження") {
      якщо опція.значення.заповнено {
        вихід_місцезнаходжень = опція.значення.значення;
        визначено_вихід_місцезнаходжень = так;
      } інакше {
       біб::надрукувати("Опція виходу --місцезнаходження має мати значення.");
       вернути 1;
     }
    } інакше {
      біб::вивести("Неочікувана опція виходу ");
      біб::надрукувати(опція.ключ);
    }

    п += 1;
  }

  р = біб::шлях_з_тексту(вхід, шлях_входу::адреса);
  якщо р < 0 {
    біб::вивести("Не вдалось створити шлях для файлу ");
    біб::надрукувати(вхід);
    статус = 1;
    стрибнути завершення;
  }

  р = біб::прочитати_файл_за_шляхом(шлях_входу, байти_входу::адреса);
  якщо р < 0 {
    біб::вивести("Не вдалось прочитати файл ");
    біб::надрукувати(вхід);
    статус = 1;
    стрибнути завершення;
  }

  змінна позиція_помилки: природне = 0;
  змінна місцезнаходження: біб::МісцезнаходженняПерекодованого;

  якщо не біб::перекодувати_з_ю8(ю8 { байти_входу.дані, байти_входу.розмір }, ні, так, значення_виходу::адреса, місцезнаходження::адреса, позиція_помилки::адреса) {
    біб::вивести("Не вдалось перекодувати файл ");
    біб::надрукувати(вхід);
    статус = 1;
    стрибнути завершення;
  }

  якщо вихід.розмір == 0 {
    біб::вивести_байти(значення_виходу.розмір, значення_виходу.дані);
  } інакше {
    р = біб::шлях_з_тексту(вихід, шлях_виходу::адреса);
    якщо р < 0 {
      біб::вивести("Не вдалось створити шлях для файлу ");
      біб::надрукувати(вихід);
      статус = 1;
      стрибнути завершення;
    }

    р = біб::перезаписати_файл_за_шляхом(шлях_виходу, значення_виходу.розмір, значення_виходу.дані);
    якщо р < 0 {
      біб::вивести("Не вдалось записати файл ");
      біб::надрукувати(вихід);
      статус = 1;
      стрибнути завершення;
    }
  }

  якщо визначено_вихід_місцезнаходжень {
    р = біб::шлях_з_тексту(вихід_місцезнаходжень, шлях_виходу_місцезнаходжень::адреса);
    якщо р < 0 {
      біб::вивести("Не вдалось створити шлях для файлу ");
      біб::надрукувати(вихід);
      статус = 1;
      стрибнути завершення;
    }

    р = біб::перезаписати_файл_за_шляхом(шлях_виходу_місцезнаходжень, місцезнаходження.розмір * 4, місцезнаходження.дані як *п8);
    якщо р < 0 {
      біб::вивести("Не вдалось записати файл ");
      біб::надрукувати(вихід);
      статус = 1;
      стрибнути завершення;
    }
  }

завершення:
  якщо треба_знищити_залишки {
    біб::знищити_шлях(шлях_входу);
    біб::знищити_шлях(шлях_виходу);
    біб::знищити_шлях(шлях_виходу_місцезнаходжень);
    біб::звільнити(байти_входу.дані);
    біб::звільнити(місцезнаходження.дані);
  }

  вернути статус;
}
