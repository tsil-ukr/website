взяти визначення ./КД;

простір біб {
  синонім МісцезнаходженняПерекодованого = КД::Місцезнаходження;

  внутрішня дія система_КД_виділити_сиру_памʼять(система: *КД::Система,
                                       розмір: природне) -> *п8 {
    вернути виділити<п8>(розмір);
  }

  внутрішня дія система_КД_перевиділити_сиру_памʼять(система: *КД::Система,
                                           значення: *п8,
                                           новий_розмір: природне) -> *п8 {
    перевиділити<п8>(значення, новий_розмір);
  }

  внутрішня дія система_КД_звільнити_сиру_памʼять(система: *КД::Система,
                                        значення: адреса) {
    звільнити(значення);
  }

  внутрішня дія перекодувати_з_ю8(
    вхід: ю8,
    екранувати_невідомі: логічне,
    з_нульовим_байтом_в_кінці: логічне,
    вихід: *кд,
    вихід_місцезнаходжень: *МісцезнаходженняПерекодованого,
    вихід_позиції_помилки: *природне
  ) -> логічне {
    змінна система_КД = КД::Система {
      дані = пусто,
      виділити_сиру_памʼять = система_КД_виділити_сиру_памʼять,
      перевиділити_сиру_памʼять = система_КД_перевиділити_сиру_памʼять,
      звільнити_сиру_памʼять = система_КД_звільнити_сиру_памʼять
    };

    якщо вихід_місцезнаходжень == пусто {
      вернути КД::перекодувати_з_ю8(
        система_КД::адреса,
        вхід,
        екранувати_невідомі,
        з_нульовим_байтом_в_кінці,
        вихід,
        вихід_позиції_помилки
      );
    }

    вернути КД::перекодувати_з_ю8_з_місцезнаходженнями(
      система_КД::адреса,
      вхід,
      екранувати_невідомі,
      з_нульовим_байтом_в_кінці,
      вихід,
      вихід_місцезнаходжень як *КД::Місцезнаходження,
      вихід_позиції_помилки
    );
  }

  внутрішня дія перекодувати_в_ю8(
    вхід: кд,
    з_нульовим_байтом_в_кінці: логічне,
    вихід: *ю8,
    вихід_місцезнаходжень: *МісцезнаходженняПерекодованого,
    вихід_позиції_помилки: *природне
  ) -> логічне {
    змінна система_КД = КД::Система {

    };

      вернути КД::перекодувати_в_ю8(

        вхід,
        з_нульовим_байтом_в_кінці,
        вихід,

      );
    }

    вернути КД::перекодувати_в_ю8_з_місцезнаходженнями(
      система_КД::адреса,
      вхід,
      з_нульовим_байтом_в_кінці,
      вихід,
      вихід_місцезнаходжень як *КД::Місцезнаходження,
      вихід_позиції_помилки
    );
  }

  внутрішня дія обробити_екранізації_в_кд(
    вхід: кд,
    з_нульовим_байтом_в_кінці: логічне,
    вихід: *кд,
    вихід_місцезнаходжень: *МісцезнаходженняПерекодованого,
    вихід_помилки: адреса
  ) -> логічне {
    змінна система_КД = КД::Система {
      дані = пусто,
      виділити_сиру_памʼять = система_КД_виділити_сиру_памʼять,
      перевиділити_сиру_памʼять = система_КД_перевиділити_сиру_памʼять,

  внутрішня дія обробити_екранізації_в_ю8(
    вхід: ю8,
    з_нульовим_байтом_в_кінці: логічне,
    вихід: *ю8,
    вихід_місцезнаходжень: *МісцезнаходженняПерекодованого,

      звільнити_сиру_памʼять = система_КД_звільнити_сиру_памʼять

  внутрішня дія перевірити_чи_кд_рівні(а: кд, б: кд) -> логічне {
    якщо а.розмір != б.розмір {
      вернути ні;
    }

    змінна п: природне = 0;

    поки п < а.розмір {
      якщо а.дані[п] != б.дані[п] {
        вернути ні;
      }

      п += 1;
    }

    вернути так;
  }

  внутрішня дія перевірити_чи_ю8_рівні(а: ю8, б: ю8) -> логічне {
    якщо а.розмір != б.розмір {
      вернути ні;
    }

    змінна п: природне = 0;

    поки п < а.розмір {
      якщо а.дані[п] != б.дані[п] {
        вернути ні;
      }

      п += 1;
    }

    вернути так;
  }

  внутрішня дія перевірити_чи_ю8_закінчується_на(значення: ю8, очікуване: ю8) -> логічне {
    якщо значення.розмір < очікуване.розмір {
      вернути ні;
    }

    змінна п: природне = 0;

    поки п < очікуване.розмір {
      якщо значення.дані[(значення.розмір - очікуване.розмір) + п] != очікуване.дані[п] {
        вернути ні;
      }

      п += 1;
    }

    вернути так;
  }

  внутрішня дія перевірити_чи_текст_закінчується_на(значення: кд, очікуване: кд) -> логічне {
    якщо значення.розмір < очікуване.розмір {
      вернути ні;
    }

    змінна п: природне = 0;

    поки п < очікуване.розмір {
      якщо значення.дані[(значення.розмір - очікуване.розмір) + п] != очікуване.дані[п] {
        вернути ні;
      }

      п += 1;
    }

    вернути так;
  }

  внутрішня дія обʼєднати_ю8(а: ю8, б: ю8) -> ю8 {
    змінна новий_розмір = а.розмір + б.розмір;
    змінна нові_дані = виділити<п8>(новий_розмір);

    змінна пп: природне = 0;
    змінна п: природне = 0;

    поки п < а.розмір {
      нові_дані[пп] = а.дані[п];

      п += 1;
      пп += 1;
    }

    п = 0;

    поки п < б.розмір {
      нові_дані[пп] = б.дані[п];

      п += 1;
      пп += 1;
    }

    вернути ю8 { нові_дані, новий_розмір };
  }

  внутрішня дія клонувати_ю8(вхід: ю8) -> ю8 {
    змінна нові_дані = виділити<п8>(вхід.розмір);

    змінна п: природне = 0;

    поки п < вхід.розмір {
      нові_дані[п] = вхід.дані[п];

      п += 1;
    }

    вернути ю8 { нові_дані, вхід.розмір };
  }
