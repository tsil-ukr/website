  внутрішня дія знищити_готовий_аргумент_шаблону(компілятор: *Компілятор, готовий_аргумент_шаблону: ГотовийАргументШаблону) {
  }

  дія знищити_готові_аргументи_шаблону(компілятор: *Компілятор, готові_аргументи_шаблону: ГотовіАргументиШаблону) {
    змінна п: природне = 0;

    поки п < готові_аргументи_шаблону.розмір {
      знищити_готовий_аргумент_шаблону(компілятор, готові_аргументи_шаблону.дані[п]);

      п += 1;
    }

    біб::звільнити(готові_аргументи_шаблону.дані);
  }
}
