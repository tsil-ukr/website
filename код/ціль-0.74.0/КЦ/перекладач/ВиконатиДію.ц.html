взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_виконати_дію(компілятор: *Компілятор, контекст: Контекст, дані_виконати_дію: розбирач::ДаніСполукиВиконатиДію, місцезнаходження: розбирач::Місцезнаходження, очікуваний_тип: *ОбʼєктТипу) -> РезультатОбʼєкта {
    змінна обʼєкт: Обʼєкт;
    змінна перший_аргумент: Обʼєкт;
    змінна є_перший_аргумент = ні;

    якщо дані_виконати_дію.ліво.вид == розбирач::ВидСполукиОтриматиВластивість {
      змінна дані_ов = дані_виконати_дію.ліво.дані як розбирач::ДаніСполукиОтриматиВластивість;

      змінна результат_ліво = скомпілювати_значення(компілятор, контекст, пусто, дані_ов.ліво, пусто);
      якщо результат_ліво.помилка != пусто {
        вернути РезультатОбʼєкта { помилка = результат_ліво.помилка, ... };
      }

      змінна назва_кд = кд { пусто, 0 };

      якщо не розбирач::отримати_значення_ідентифікатора(
        контекст.розібраний_текст.текст_коду.значення,
        контекст.розібраний_текст.слова,
        дані_ов.ідентифікатор,
        назва_кд::адреса
      ) {
        вернути РезультатОбʼєкта {
          помилка = створити_помилку(
            компілятор,
            "Не вдалось отримати назву властивості.",
            створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
          ),
          ...
        };
      }

      змінна назва = отримати_назву(компілятор, назва_кд);

      змінна результат_власт = обʼєкт_отримати_властивість(
        компілятор,
        контекст,
        дані_виконати_дію.ліво.місцезнаходження,
        результат_ліво.обʼєкт,
        назва,
        пусто
      );
      якщо результат_власт.помилка != пусто {
        вернути РезультатОбʼєкта { помилка = результат_власт.помилка, ... };
      }

      обʼєкт = результат_власт.обʼєкт;

      змінна тип_значення: *ОбʼєктТипу = пусто;

      якщо обʼєкт.вид == ВидОбʼєктаДія {
        змінна д = обʼєкт.дані як *ОбʼєктДії;

        тип_значення = д.тип;
      } інакше якщо обʼєкт.вид == ВидОбʼєктаНезавантаженеЗначення {
        змінна д = обʼєкт.дані як ОбʼєктНезавантаженогоЗначення;

        тип_значення = д.тип;
      } інакше якщо обʼєкт.вид == ВидОбʼєктаЗначення {
        змінна д = обʼєкт.дані як ОбʼєктЗначення;

        тип_значення = д.тип;
      }

      якщо тип_значення != пусто {
        якщо тип_значення.вид == ВидОбʼєктаТипуДія {
          змінна д = тип_значення.дані як ДаніОбʼєктаТипуДія;

          якщо д.параметри.розмір > 0 {
            змінна п = д.параметри.дані[0];

            якщо п.назва == отримати_назву(компілятор, "я") {
              змінна тп = п.тип;
              змінна ол = результат_ліво.обʼєкт;

              якщо ол.вид == ВидОбʼєктаНезавантаженеЗначення {
                змінна нз = ол.дані як ОбʼєктНезавантаженогоЗначення;

                змінна новий_обʼєкт_значення: ОбʼєктЗначення;

                якщо нз.тип.вид == ВидОбʼєктаТипуАдреса {
                  новий_обʼєкт_значення.тип = нз.тип;
                  новий_обʼєкт_значення.пм_значення = ПМ::вказівка_прочитати(контекст.кк_дії.поточний_крок.пм_крок, нз.тип.пм_тип, нз.пм_значення);
                } інакше {
                  новий_обʼєкт_значення.тип = створити_тип_адреси(компілятор, нз.тип);
                  новий_обʼєкт_значення.пм_значення = нз.пм_значення;
                }

                перший_аргумент = Обʼєкт { вид = ВидОбʼєктаЗначення, дані = новий_обʼєкт_значення };
                є_перший_аргумент = так;
              }
            }
          }
        }
      }
    } інакше {
      змінна результат_ліво = скомпілювати_значення(компілятор, контекст, пусто, дані_виконати_дію.ліво, пусто);
      якщо результат_ліво.помилка != пусто {
        вернути РезультатОбʼєкта { помилка = результат_ліво.помилка, ... };
      }

      обʼєкт = результат_ліво.обʼєкт;
    }

    змінна зсув_аргументів = природне(0);

    якщо є_перший_аргумент {
      зсув_аргументів = 1;
    }

    змінна аргументи = біб::виділити<Аргумент>(
      дані_виконати_дію.аргументи.розмір + зсув_аргументів
    );

    якщо є_перший_аргумент {
      аргументи[0] = Аргумент {
        обʼєкт = перший_аргумент,
        місцезнаходження = місцезнаходження
      };
    }

    змінна п: природне = 0;
    поки п < дані_виконати_дію.аргументи.розмір {
      змінна результат_аргумента = скомпілювати_значення(
        компілятор,
        контекст,
        пусто,
        дані_виконати_дію.аргументи.дані[п].значення,
        пусто
      );

      якщо результат_аргумента.помилка != пусто {
        біб::звільнити(аргументи);

        вернути РезультатОбʼєкта { помилка = результат_аргумента.помилка, ... };
      }

      аргументи[п + зсув_аргументів] = Аргумент {
        обʼєкт = результат_аргумента.обʼєкт,
        місцезнаходження = дані_виконати_дію.аргументи.дані[п].значення.місцезнаходження
      };

      п += 1;
    }

    змінна результат = обʼєкт_виконати(
      компілятор,
      контекст,
      місцезнаходження,
      обʼєкт,
      дані_виконати_дію.аргументи.розмір + зсув_аргументів,
      аргументи
    );

    біб::звільнити(аргументи);

    вернути результат;
  }
}
