взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_отримати_секційну_властивість(компілятор: *Компілятор, контекст: Контекст, дані_отримати_секційну_властивість: розбирач::ДаніСполукиОтриматиСекційнуВластивість, місцезнаходження: розбирач::Місцезнаходження, очікуваний_тип: *ОбʼєктТипу) -> РезультатОбʼєкта {
    змінна результат_ліво = скомпілювати_значення(компілятор, контекст, пусто, дані_отримати_секційну_властивість.ліво, пусто);
    якщо результат_ліво.помилка != пусто {
      вернути РезультатОбʼєкта { ..., помилка = результат_ліво.помилка };
    }

    змінна назва_кд = кд { пусто, 0 };

    якщо не розбирач::отримати_значення_ідентифікатора(
      контекст.розібраний_текст.текст_коду.значення,
      контекст.розібраний_текст.слова,
      дані_отримати_секційну_властивість.ідентифікатор,
      назва_кд::адреса
    ) {
      вернути РезультатОбʼєкта {
        помилка = створити_помилку(
          компілятор,
          "Не вдалось отримати назву властивості секції.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        ),
        ...
      };
    }

    змінна назва = отримати_назву(компілятор, назва_кд);

    змінна результат = обʼєкт_отримати_властивість_секції(
      компілятор,
      контекст,
      місцезнаходження,
      результат_ліво.обʼєкт,
      назва,
      очікуваний_тип
    );

    вернути результат;
  }
}
