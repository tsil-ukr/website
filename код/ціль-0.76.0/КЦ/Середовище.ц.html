простір КЦ {
  внутрішня дія перед_заміною_субʼєкта(реєстр: *біб::Реєстр<*Назва, Обʼєкт>, ак: *Назва, аз: Обʼєкт, бк: **Назва, бз: *Обʼєкт) {
  }

  внутрішня дія перед_видаленням_субʼєкта(реєстр: *біб::Реєстр<*Назва, Обʼєкт>, ключ: *Назва, значення: Обʼєкт) {
  }

  дія створити_середовище(компілятор: *Компілятор, зовнішнє: *Середовище) -> *Середовище {
    змінна середовище = біб::виділити<Середовище>(1);

    середовище.наступне = пусто;
    середовище.компілятор = компілятор;
    середовище.зовнішнє = зовнішнє;
    середовище.субʼєкти = біб::виділити_реєстр<*Назва, Обʼєкт>(
      0,
      компілятор,
      змішати_назву_шлях_для_реєстру<Обʼєкт>,
      перевірити_назви_з_реєстру<Обʼєкт>,
      перед_заміною_субʼєкта,
      перед_видаленням_субʼєкта
    );
    середовище.обʼєкт_секції = пусто;
    середовище.обʼєкт_дії = пусто;

    якщо компілятор.перше_середовище == пусто {
      компілятор.перше_середовище = середовище;
    }
    якщо компілятор.останнє_середовище == пусто {
      компілятор.останнє_середовище = середовище;
    } інакше {
      компілятор.останнє_середовище.наступне = середовище;
      компілятор.останнє_середовище = середовище;
    }

    вернути середовище;
  }

  дія створити_середовище_секції(компілятор: *Компілятор, зовнішнє: *Середовище, обʼєкт_секції: *ОбʼєктСекції) -> *Середовище {
    змінна середовище = створити_середовище(компілятор, зовнішнє);
    середовище.обʼєкт_секції = обʼєкт_секції;
    вернути середовище;
  }

  дія створити_середовище_дії(компілятор: *Компілятор, зовнішнє: *Середовище, обʼєкт_дії: *ОбʼєктДії) -> *Середовище {
    змінна середовище = створити_середовище(компілятор, зовнішнє);
    середовище.обʼєкт_дії = обʼєкт_дії;
    вернути середовище;
  }

  дія знищити_середовище(середовище: *Середовище) {
    змінна компілятор = середовище.компілятор;

    біб::звільнити_реєстр<*Назва, Обʼєкт>(середовище.субʼєкти);

    біб::звільнити(середовище);
  }

  дія змінити_в_середовищі(середовище: *Середовище, назва: *Назва, обʼєкт: Обʼєкт) {
    біб::змінити_в_реєстрі<*Назва, Обʼєкт>(середовище.субʼєкти, назва, обʼєкт);
  }

  дія знайти_в_середовищі(середовище: *Середовище, назва: *Назва, вихід: *Обʼєкт) -> логічне {
    вернути біб::знайти_в_реєстрі<*Назва, Обʼєкт>(середовище.субʼєкти, назва, вихід);
  }

  дія знайти_в_середовищі_рекурсивно(середовище: *Середовище, назва: *Назва, вихід: *Обʼєкт) -> логічне {
    змінна поточне_середовище = середовище;

    поки поточне_середовище != пусто {
      якщо біб::знайти_в_реєстрі<*Назва, Обʼєкт>(поточне_середовище.субʼєкти, назва, вихід) {
        вернути так;
      }

      поточне_середовище = поточне_середовище.зовнішнє;
    }

    вернути ні;
  }
}
