взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_синонім(компілятор: *Компілятор, контекст: Контекст, контекст_виконання_шаблону: *КонтекстКомпіляціїВиконанняШаблону, дані_синоніма: розбирач::ДаніСполукиСинонім, місцезнаходження: розбирач::Місцезнаходження) -> РезультатОбʼєкта {
    вернути скомпілювати_значення(
      компілятор,
      контекст,
      пусто,
      дані_синоніма.значення,
      пусто
    );
  }

  дія скомпілювати_визначити_синонім(компілятор: *Компілятор, контекст: Контекст, дані_синоніма: розбирач::ДаніСполукиСинонім, місцезнаходження: розбирач::Місцезнаходження) -> РезультатОбʼєкта {
    змінна назва_кд = кд { пусто, 0 };

    якщо не розбирач::отримати_значення_слова(
      контекст.розібраний_текст.текст_коду.значення,
      контекст.розібраний_текст.слова.дані[дані_синоніма.ідентифікатор.позиція_слова],
      назва_кд::адреса
    ) {
      вернути РезультатОбʼєкта {
        помилка = створити_помилку(
          компілятор,
          "Не вдалось отримати назву змінної.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        ),
        ...
      };
    }

    змінна назва = отримати_назву(компілятор, назва_кд);

    змінна обʼєкт_з_середовища: Обʼєкт;
    якщо знайти_в_середовищі(контекст.середовище, назва, обʼєкт_з_середовища::адреса) {
      вернути РезультатОбʼєкта {
        ...,
        помилка = створити_помилку(
          компілятор,
          "Субʼєкт вже визначено.",
          створити_місцезнаходження_помилки_з_позиції_слова(компілятор, контекст.розібраний_текст, дані_синоніма.ідентифікатор.позиція_слова)
        )
      };
    }

    змінна результат = скомпілювати_синонім(
      компілятор,
      контекст,
      пусто,
      дані_синоніма,
      місцезнаходження
    );

    якщо результат.помилка != пусто {
      вернути результат;
    }

    змінити_в_середовищі(контекст.середовище, назва, результат.обʼєкт);

    вернути результат;
  }
}
