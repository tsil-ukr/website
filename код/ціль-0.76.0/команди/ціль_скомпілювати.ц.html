дія знайти_границі_рядка(значення: ю8, позиція_в_тексті: природне, вихід_початку: *природне, вихід_кінця: *природне) {
  змінна початок = позиція_в_тексті;
  поки початок > 0 {
    якщо значення.дані[початок - 1] == 10 {
      стрибнути далі_кінець;
    }
    початок -= 1;
  }
далі_кінець:
  змінна кінець = позиція_в_тексті;
  поки кінець < значення.розмір {
    якщо значення.дані[кінець] == 10 {
      стрибнути вихід;
    }
    кінець += 1;
  }
вихід:
  вихід_початку::вміст = початок;
  вихід_кінця::вміст = кінець;
}

дія порахувати_кількість_символів_ю8(значення: ю8) -> природне {
  змінна кількість: природне = 0;
  змінна п: природне = 0;
  поки п < значення.розмір {
    змінна с = значення.дані[п];
    якщо с < 128 {
      п += 1;
    } інакше якщо (с & 224) == 192 {
      п += 2;
    } інакше якщо (с & 240) == 224 {
      п += 3;
    } інакше якщо (с & 248) == 240 {
      п += 4;
    } інакше {
      вернути кількість;
    }
    кількість += 1;
  }
  вернути кількість;
}

дія порахувати_зсув_ю8(значення: ю8, позиція_символа: природне) -> природне {
  змінна п: природне = 0;
  змінна символи: природне = 0;
  поки п < значення.розмір {
    якщо символи == позиція_символа {
      вернути п;
    }
    змінна с = значення.дані[п];
    якщо с < 128 {
      п += 1;
    } інакше якщо (с & 224) == 192 {
      п += 2;
    } інакше якщо (с & 240) == 224 {
      п += 3;
    } інакше якщо (с & 248) == 240 {
      п += 4;
    } інакше {
      вернути п;
    }
    символи += 1;
  }
  вернути п;
}

дія виправити_ю8_якщо_незакінчено(значення: ю8) -> ю8 {
  змінна п: природне = 0;
  поки п < значення.розмір {
    змінна с = значення.дані[п];
    якщо с < 128 {
      п += 1;
    } інакше якщо (с & 224) == 192 {
      якщо п + 1 < значення.розмір {
        п += 2;
      } інакше {
        значення.розмір = значення.розмір - 1;
        вернути значення;
      }
    } інакше якщо (с & 240) == 224 {
      якщо п + 2 < значення.розмір {
        п += 3;
      } інакше {
        значення.розмір = значення.розмір - 2;
        вернути значення;
      }
    } інакше якщо (с & 248) == 240 {
      якщо п + 3 < значення.розмір {
        п += 4;
      } інакше {
        значення.розмір = значення.розмір - 3;
        вернути значення;
      }
    } інакше {
      вернути значення;
    }
  }
  вернути значення;
}

дія витнути_значення_ю8(значення: ю8, позиція_помилки: природне, максимальна_довжина: природне) -> ВитнутеЗначенняЗіСловомЮ8 {
  змінна результат = ВитнутеЗначенняЗіСловомЮ8 {
    витнуте_значення = ю8 { пусто, 0 },
    початок_підкреслення = 0
  };
  якщо максимальна_довжина == 0 {
    вернути результат;
  }
  змінна рядок_початок: природне = 0;
  змінна рядок_кінець: природне = 0;
  знайти_границі_рядка_ю8(значення, позиція_помилки, рядок_початок::адреса, рядок_кінець::адреса);
  змінна рядок = ю8 {
    значення.дані[рядок_початок]::адреса,
    рядок_кінець - рядок_початок,
  };
  змінна позиція_символа_помилки = порахувати_кількість_символів_ю8(ю8 { рядок.дані, позиція_помилки - рядок_початок });
  змінна рядок_символи = порахувати_кількість_символів_ю8(рядок);
  змінна макс_байти = максимальна_довжина;
  якщо макс_байти > рядок.розмір {
    макс_байти = рядок.розмір;
  }
  змінна половина_символів = макс_байти / 2;
  змінна під_початок_символ: природне = 0;
  якщо позиція_символа_помилки > половина_символів {
    під_початок_символ = позиція_символа_помилки - половина_символів;
  }
  якщо (під_початок_символ + макс_байти) > рядок_символи {
    якщо рядок_символи > макс_байти {
      під_початок_символ = рядок_символи - макс_байти;
    } інакше {
      під_початок_символ = 0;
    }
  }
  змінна під_початок_байт = порахувати_зсув_ю8(рядок, під_початок_символ);
  змінна під_кінець_байт = під_початок_байт + макс_байти;
  якщо під_кінець_байт > рядок.розмір {
    під_кінець_байт = рядок.розмір;
  }
  змінна під_розмір = під_кінець_байт - під_початок_байт;
  результат.витнуте_значення = виправити_ю8_якщо_незакінчено(
    ю8 {
      рядок.дані[під_початок_байт]::адреса,
      під_розмір,
    }
  );
  якщо позиція_символа_помилки >= під_початок_символ {
    результат.початок_підкреслення = позиція_символа_помилки - під_початок_символ;
  } інакше {
    результат.початок_підкреслення = 0;
  }
  вернути результат;
}

дія ціль_скомпілювати(
  опції_програми: цілеспрямовано::Опції<кд>,
  опції_команди: цілеспрямовано::Опції<кд>,
  вхід: кд,
  опції_входу: цілеспрямовано::Опції<кд>,
  вихід: кд,
  опції_виходу: цілеспрямовано::Опції<кд>,
  треба_знищити_залишки: логічне
) -> ц32 {
  змінна р: ціле = 0;

  змінна формат_виходу: природне = ФОРМАТ_ВИХОДУ_ЛЛВМО;
  змінна визначено_формат_виходу = ні;

  змінна пм_платформа_виходу: природне = ПМ::ПЛАТФОРМА_ІКС86_64_ПК_ЛІНУКС_ГНУ;

  змінна шлях_до_біб = кд { пусто, 0 };

  змінна п: природне = 0;

  поки п < опції_команди.розмір {
    змінна опція = опції_команди.дані[п];

    якщо не опція.коротка і біб::перевірити_чи_кд_рівні(опція.ключ, "формат-виходів") {
      якщо опція.значення.заповнено {
        якщо розібрати_значення_опції_формат_виходу(опція.значення.значення, "--формат-виходів", так, формат_виходу::адреса) {
          визначено_формат_виходу = так;
        } інакше {
          вернути 1;
        }
      } інакше {
       біб::надрукувати("Опція команди --формат-виходів має мати значення.");
       вернути 1;
     }
    } інакше якщо не опція.коротка і біб::перевірити_чи_кд_рівні(опція.ключ, "платформа-виходів") {
      якщо опція.значення.заповнено {
        якщо розібрати_значення_опції_платформа_виходу(опція.значення.значення, "--платформа-виходів", так, пм_платформа_виходу::адреса) {
        } інакше {
          вернути 1;
        }
      } інакше {
       біб::надрукувати("Опція команди --платформа-виходів має мати значення.");
       вернути 1;
      }
    } інакше якщо не опція.коротка і (біб::перевірити_чи_кд_рівні(опція.ключ, "бібліотека-входів") або біб::перевірити_чи_кд_рівні(опція.ключ, "біб-входів")) {
      якщо опція.значення.заповнено {
        шлях_до_біб = опція.значення.значення;
      } інакше {
       біб::надрукувати("Опція команди --бібліотека-входів має мати значення.");
       вернути 1;
      }
    } інакше {
      біб::вивести("Неочікувана опція команди ");
      біб::надрукувати(опція.ключ);
    }

    п += 1;
  }

  змінна шляхи_до_визначень = біб::зробити_накопичувач<кд>(0);

  п = 0;

  поки п < опції_входу.розмір {
    змінна опція = опції_входу.дані[п];

    якщо опція.коротка і біб::перевірити_чи_кд_рівні(опція.ключ, "Б") {
      якщо опція.значення.заповнено {
        шлях_до_біб = опція.значення.значення;
      } інакше {
         біб::надрукувати("Опція входу -Б має мати значення.");
         вернути 1;
      }
    } інакше якщо не опція.коротка і (біб::перевірити_чи_кд_рівні(опція.ключ, "бібліотека") або біб::перевірити_чи_кд_рівні(опція.ключ, "біб")) {
      якщо опція.значення.заповнено {
        шлях_до_біб = опція.значення.значення;
      } інакше {
         біб::надрукувати("Опція входу --бібліотека має мати значення.");
         вернути 1;
      }
    } інакше якщо опція.коротка і біб::перевірити_чи_кд_рівні(опція.ключ, "В") {
      якщо опція.значення.заповнено {
        біб::покласти_в_накопичувач<кд>(шляхи_до_визначень::адреса, опція.значення.значення);
      } інакше {
         біб::надрукувати("Опція входу -В має мати значення.");
         вернути 1;
      }
    } інакше якщо не опція.коротка і біб::перевірити_чи_кд_рівні(опція.ключ, "визначення") {
      якщо опція.значення.заповнено {
        біб::покласти_в_накопичувач<кд>(шляхи_до_визначень::адреса, опція.значення.значення);
      } інакше {
         біб::надрукувати("Опція входу --визначення має мати значення.");
         вернути 1;
      }
    } інакше {
      біб::вивести("Неочікувана опція входу ");
      біб::надрукувати(опція.ключ);
    }

    п += 1;
  }

  п = 0;

  поки п < опції_виходу.розмір {
    змінна опція = опції_виходу.дані[п];

    якщо опція.коротка і біб::перевірити_чи_кд_рівні(опція.ключ, "Ф") {
      якщо опція.значення.заповнено {
        якщо розібрати_значення_опції_формат_виходу(опція.значення.значення, "-Ф", ні, формат_виходу::адреса) {
          визначено_формат_виходу = так;
        } інакше {
          вернути 1;
        }
      } інакше {
       біб::надрукувати("Опція виходу -Ф має мати значення.");
       вернути 1;
     }
    } інакше якщо не опція.коротка і біб::перевірити_чи_кд_рівні(опція.ключ, "формат") {
      якщо опція.значення.заповнено {
        якщо розібрати_значення_опції_формат_виходу(опція.значення.значення, "--формат", ні, формат_виходу::адреса) {
          визначено_формат_виходу = так;
        } інакше {
          вернути 1;
        }
      } інакше {
       біб::надрукувати("Опція виходу --формат має мати значення.");
       вернути 1;
     }
    } інакше якщо опція.коротка і біб::перевірити_чи_кд_рівні(опція.ключ, "П") {
      якщо опція.значення.заповнено {
        якщо розібрати_значення_опції_платформа_виходу(опція.значення.значення, "-П", ні, пм_платформа_виходу::адреса) {
        } інакше {
          вернути 1;
        }
      } інакше {
       біб::надрукувати("Опція виходу -П має мати значення.");
       вернути 1;
     }
    } інакше якщо не опція.коротка і біб::перевірити_чи_кд_рівні(опція.ключ, "платформа") {
      якщо опція.значення.заповнено {
        якщо розібрати_значення_опції_платформа_виходу(опція.значення.значення, "--платформа", ні, пм_платформа_виходу::адреса) {
        } інакше {
          вернути 1;
        }
      } інакше {
       біб::надрукувати("Опція виходу --платформа має мати значення.");
       вернути 1;
     }
    } інакше {
      біб::вивести("Неочікувана опція виходу ");
      біб::надрукувати(опція.ключ);
    }

    п += 1;
  }

  якщо не визначено_формат_виходу {
    якщо очікується_ЛЛВМІР(вихід) {
      формат_виходу = ФОРМАТ_ВИХОДУ_ЛЛВМІР;
    } інакше якщо очікується_ЛЛВМО(вихід) {
      формат_виходу = ФОРМАТ_ВИХОДУ_ЛЛВМО;
    }
  }

  змінна пм_модуль = ПМ::створити_модуль(
    пусто,
    пм_платформа_виходу
  );

  змінна компілятор = КЦ::створити_компілятор(пм_модуль);

  п = 0;

  поки п < шляхи_до_визначень.розмір {
    змінна шлях_до_визначення = шляхи_до_визначень.дані[п];

    змінна шлях_ю8: ю8;
    змінна шлях: біб::Шлях;

    біб::перекодувати_в_ю8(шлях_до_визначення, так, шлях_ю8::адреса, пусто, пусто);

    біб::шлях_з_ю8(шлях_ю8, шлях::адреса);

    біб::звільнити(шлях_ю8.дані);

    біб::покласти_в_накопичувач<біб::Шлях>(компілятор.шляхи_до_визначень::адреса, шлях);

    п += 1;
  }

  якщо біб::отримати_робочу_директорію_процесу(компілятор.кореневий_шлях::адреса) < 0 {
    КЦ::померти(компілятор, "Не вдалось отримати поточну директорію процесу.");
  }

  біб::шлях_з_тексту(шлях_до_біб, компілятор.шлях_до_біб::адреса);

  змінна помилка: *КЦ::Помилка = пусто;

  змінна шлях_входу: біб::Шлях;
  біб::шлях_з_тексту(вхід, шлях_входу::адреса);

  змінна успіх_взяття = КЦ::взяти_файл_ю8(
    компілятор,
    компілятор.глобальне_середовище,
    шлях_входу,
    помилка::адреса
  );

  якщо не успіх_взяття {
    якщо помилка.місцезнаходження != пусто {
      змінна текст_коду = помилка.місцезнаходження.текст_коду;
      біб::вивести_ю8(ю8 { текст_коду.шлях.дані, текст_коду.шлях.розмір });
      біб::вивести(":");
      біб::вивести_природне(помилка.місцезнаходження.рядок);
      біб::вивести(":");
      біб::вивести_природне(помилка.місцезнаходження.стовпець);
      біб::вивести(" помилка: ");
    }

    змінна повідомлення_ю8 = ю8 { пусто, 0 };
    змінна позиція_помилки: природне = 0;
    якщо не біб::перекодувати_в_ю8(помилка.повідомлення, так, повідомлення_ю8::адреса, пусто, позиція_помилки::адреса) {
      КЦ::померти(компілятор, "Не вдалось перекодувати помилку повідомлення в ю8.");
    }
    біб::надрукувати_ю8(повідомлення_ю8);
    біб::звільнити(повідомлення_ю8.дані);

    якщо помилка.місцезнаходження != пусто {
      змінна текст_коду = помилка.місцезнаходження.текст_коду;
      змінна оригінальне_значення = текст_коду.оригінальне_значення;

      змінна витнутний_текст = витнути_значення_ю8(
        оригінальне_значення.значення,
        природне(оригінальне_значення.місцезнаходження.дані[помилка.місцезнаходження.позиція_в_тексті]),
        100
      );

      змінна природне_ю8 = біб::перетворити_природне_на_ю8(помилка.місцезнаходження.рядок);
      біб::вивести_ю8(природне_ю8);
      біб::вивести("| ");
      біб::надрукувати_ю8(витнутний_текст.витнуте_значення);
      змінна п: природне = 0;
      поки п < (витнутний_текст.початок_підкреслення + природне_ю8.розмір + 2) {
        біб::вивести(" ");
        п += 1;
      }
      біб::надрукувати("^");
      біб::звільнити(природне_ю8.дані);
    }

    якщо треба_знищити_залишки {
      КЦ::знищити_помилку(компілятор, помилка);
      КЦ::знищити_компілятор(компілятор);
    }

    вернути 1;
  }

  змінна значення_запису = КЦ::Байти { 0, пусто };

  якщо формат_виходу == ФОРМАТ_ВИХОДУ_ЛЛВМІР {
    ПМЛЛВМ::перетворити_на_ллвмір(пм_модуль, значення_запису.розмір::адреса, значення_запису.дані::адреса);
  } інакше якщо формат_виходу == ФОРМАТ_ВИХОДУ_ЛЛВМО {
    // потім: помилка
  }

  якщо вихід.розмір == 0 {
    біб::вивести_байти(значення_запису.розмір, значення_запису.дані);
  } інакше {
    змінна шлях_виходу: біб::Шлях;

    р = біб::шлях_з_тексту(вихід, шлях_виходу::адреса);
    якщо р < 0 {
      біб::вивести("Не вдалось створити шлях для вихідного файлу ");
      біб::надрукувати(вихід);
    }

    р = біб::перезаписати_файл_за_шляхом(шлях_виходу, значення_запису.розмір, значення_запису.дані);
    якщо р < 0 {
      біб::вивести("Не вдалось записати у файл ");
      біб::надрукувати(вихід);
    }

    біб::знищити_шлях(шлях_виходу);
  }

  якщо треба_знищити_залишки {
    біб::звільнити(значення_запису.дані);
    КЦ::знищити_компілятор(компілятор);
    ПМ::знищити_модуль(пм_модуль);
  }

  вернути 0;
}
