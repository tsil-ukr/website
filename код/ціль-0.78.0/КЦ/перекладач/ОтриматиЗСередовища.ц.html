взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_отримати_з_середовища(компілятор: *Компілятор, контекст: Контекст, дані_прочитати: розбирач::ДаніСполукиПрочитати, місцезнаходження: розбирач::Місцезнаходження, очікуваний_тип: *ОбʼєктТипу) -> РезультатОбʼєкта {
    змінна назва_кд = кд { пусто, 0 };

    якщо не розбирач::отримати_значення_ідентифікатора(
      контекст.розібраний_текст.текст_коду.значення,
      контекст.розібраний_текст.слова,
      дані_прочитати.ідентифікатор,
      назва_кд::адреса
    ) {
      вернути РезультатОбʼєкта {
        помилка = створити_помилку(
          компілятор,
          "Не вдалось отримати назву змінної.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        ),
        ...
      };
    }

    змінна назва = отримати_назву(компілятор, назва_кд);

    змінна обʼєкт_з_середовища: Обʼєкт;
    якщо знайти_в_середовищі_рекурсивно(контекст.середовище, назва, обʼєкт_з_середовища::адреса) {
      вернути РезультатОбʼєкта { обʼєкт_з_середовища, пусто };
    }

    вернути РезультатОбʼєкта {
      помилка = створити_помилку_субʼєкт_не_визначено(
        компілятор,
        Місцезнаходження { контекст.розібраний_текст, дані_прочитати.ідентифікатор.позиція_слова },
        назва_кд
      ),
      ...
    };
  }
}
