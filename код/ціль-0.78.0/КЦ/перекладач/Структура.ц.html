взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_параметри_структури(компілятор: *Компілятор, контекст: Контекст, параметри_розбирача: розбирач::Параметри, місцезнаходження: розбирач::Місцезнаходження) -> РезультатПараметрів {
    змінна параметри = Параметри { 0, пусто, 0 };

    змінна п: природне = 0;
    поки п < параметри_розбирача.розмір {
      змінна параметр_розбирача = параметри_розбирача.дані[п];

      змінна назва_кд = кд { пусто, 0 };
      змінна позиція_слова_назви: природне = 0;

      якщо параметр_розбирача.ідентифікатор.заповнено {
        якщо не розбирач::отримати_значення_слова(
          контекст.розібраний_текст.текст_коду.значення,
          контекст.розібраний_текст.слова.дані[параметр_розбирача.ідентифікатор.значення.позиція_слова],
          назва_кд::адреса
        ) {
          знищити_параметри(компілятор, параметри);

          вернути РезультатПараметрів {
            ..., помилка = створити_помилку(
              компілятор,
              "Не вдалось отримати назву параметра структури.",
              створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
            )
          };
        }
        позиція_слова_назви = параметр_розбирача.ідентифікатор.значення.позиція_слова;
      }

      змінна назва = отримати_назву(компілятор, назва_кд);

      змінна результат_типу = скомпілювати_значення_як_тип(компілятор, контекст, параметр_розбирача.тип, так);

      якщо результат_типу.помилка != пусто {
        знищити_параметри(компілятор, параметри);

        вернути РезультатПараметрів { ..., помилка = результат_типу.помилка };
      }

      покласти_параметр(
        компілятор,
        параметри::адреса,
        Параметр {
          назва,
          результат_типу.обʼєкт_типу,
          Місцезнаходження { контекст.розібраний_текст, позиція_слова_назви },
          Місцезнаходження { контекст.розібраний_текст, параметр_розбирача.тип.місцезнаходження.позиція_початкового_слова }
        }
      );

      п += 1;
    }

    вернути РезультатПараметрів {
      параметри,
      помилка = пусто
    };
  }

  дія скомпілювати_структуру(компілятор: *Компілятор, контекст: Контекст, контекст_виконання_шаблону: *КонтекстКомпіляціїВиконанняШаблону, обʼєкт_типу_структура: *ОбʼєктТипу, дані_структури: розбирач::ДаніСполукиСтруктура, місцезнаходження: розбирач::Місцезнаходження) -> РезультатОбʼєктаТипу {
    якщо обʼєкт_типу_структура == пусто {
      змінна назва_кд = кд { пусто, 0 };

      якщо дані_структури.ідентифікатор.заповнено {
        якщо не розбирач::отримати_значення_слова(
          контекст.розібраний_текст.текст_коду.значення,
          контекст.розібраний_текст.слова.дані[дані_структури.ідентифікатор.значення.позиція_слова],
          назва_кд::адреса
        ) {
          вернути РезультатОбʼєктаТипу {
            ..., помилка = створити_помилку(
              компілятор,
              "Не вдалось отримати назву структури.",
              створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
            )
          };
        }
      }

      змінна назва = отримати_назву(компілятор, назва_кд);

      змінна повна_назва: *Назва = пусто;

      якщо назва.значення.розмір != 0 {
        повна_назва = отримати_назву(компілятор, отримати_повну_назву_в_контексті(компілятор, контекст, назва.значення));
      }

      обʼєкт_типу_структура = виділити_обʼєкт_типу(компілятор);

      змінна дії = створити_середовище(компілятор, контекст.середовище);

      змінна пакована = ні;

      змінна п: природне = 0;

      поки п < дані_структури.атрибути.розмір {
        змінна атрибут_розбирача = дані_структури.атрибути.дані[п];

        змінна назва_атрибута_кд = кд { пусто, 0 };

        якщо не розбирач::отримати_значення_слова(
          контекст.розібраний_текст.текст_коду.значення,
          контекст.розібраний_текст.слова.дані[атрибут_розбирача.ідентифікатор.позиція_слова],
          назва_атрибута_кд::адреса
        ) {
          вернути РезультатОбʼєктаТипу {
            ..., помилка = створити_помилку(
              компілятор,
              "Не вдалось отримати назву атрибута структури.",
              створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
            )
          };
        }

        якщо біб::перевірити_чи_кд_рівні(назва_атрибута_кд, "пакована") {
          пакована = так;
        }

        п += 1;
      }

      змінна дані_обʼєкта_типу_структура = ДаніОбʼєктаТипуСтруктура {
        назва = назва,
        параметри = Параметри { 0, пусто, 0 },
        повна_назва = повна_назва,
        готові_аргументи_шаблону = ГотовіАргументиШаблону { 0, пусто },
        дії = дії,
        пакована = пакована
      };

      обʼєкт_типу_структура.вид = ВидОбʼєктаТипуСтруктура;
      обʼєкт_типу_структура.дані = дані_обʼєкта_типу_структура;

      обʼєкт_типу_структура.пм_тип = ПМ::створити_структуру(
        компілятор.пм_модуль,
        повна_назва,
        ПМ::Параметри { 0, пусто, 0 },
        дані_обʼєкта_типу_структура.пакована
      );

      якщо контекст_виконання_шаблону != пусто {
        дані_обʼєкта_типу_структура.готові_аргументи_шаблону = клонувати_готові_аргументи_шаблону(компілятор, контекст_виконання_шаблону.готові_аргументи_шаблону);
        біб::змінити_в_реєстрі<ГотовіАргументиШаблону, Обʼєкт>(
          контекст_виконання_шаблону.обʼєкт_шаблону.готові,
          клонувати_готові_аргументи_шаблону(компілятор, контекст_виконання_шаблону.готові_аргументи_шаблону),
          Обʼєкт { ВидОбʼєктаТип, обʼєкт_типу_структура }
        );
        обʼєкт_типу_структура.дані = дані_обʼєкта_типу_структура;
      }

      змінити_в_середовищі(
        дії,
        отримати_назву(компілятор, "Я"),
        Обʼєкт { ВидОбʼєктаТип, обʼєкт_типу_структура }
      );

      якщо контекст_виконання_шаблону == пусто {
        якщо дані_обʼєкта_типу_структура.назва.значення.розмір != 0 {
          змінити_в_середовищі(
            дії,
            дані_обʼєкта_типу_структура.назва,
            Обʼєкт { ВидОбʼєктаТип, обʼєкт_типу_структура }
          );
        }
      }
    }

    якщо дані_структури.параметри.розмір == 0 {
      вернути РезультатОбʼєктаТипу { обʼєкт_типу_структура, помилка = пусто };
    }

    змінна дані_обʼєкта_типу_структура = обʼєкт_типу_структура.дані як ДаніОбʼєктаТипуСтруктура;

    якщо дані_обʼєкта_типу_структура.параметри.розмір != 0 {
      вернути РезультатОбʼєктаТипу {
        ...,
        помилка = створити_помилку(
          компілятор,
          "Структуру вже визначено.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        )
      };
    }

    змінна середовище_компіляції_структури = створити_середовище(компілятор, контекст.середовище);

    якщо контекст_виконання_шаблону == пусто {
      якщо дані_обʼєкта_типу_структура.назва.значення.розмір != 0 {
        змінити_в_середовищі(
          середовище_компіляції_структури,
          дані_обʼєкта_типу_структура.назва,
          Обʼєкт { ВидОбʼєктаТип, обʼєкт_типу_структура }
        );
      }
    }

    змінна результат_параметрів = скомпілювати_параметри_структури(компілятор, контекст, дані_структури.параметри, місцезнаходження);

    якщо результат_параметрів.помилка != пусто {
      вернути РезультатОбʼєктаТипу {
        ...,
        помилка = результат_параметрів.помилка
      };
    }

    дані_обʼєкта_типу_структура.параметри = результат_параметрів.параметри;

    змінна пм_параметри = ПМ::Параметри { 0, пусто, 0 };
    змінна п: природне = 0;
    поки п < дані_обʼєкта_типу_структура.параметри.розмір {
      ПМ::покласти_параметр(компілятор.пм_модуль, пм_параметри::адреса, ПМ::Параметр { дані_обʼєкта_типу_структура.параметри.дані[п].тип.пм_тип });
      п += 1;
    }

    обʼєкт_типу_структура.вид = ВидОбʼєктаТипуСтруктура;
    обʼєкт_типу_структура.дані = дані_обʼєкта_типу_структура;

    ПМ::заповнити_параметри_структури(
      обʼєкт_типу_структура.пм_тип,
      пм_параметри
    );

    обʼєкт_типу_структура.дані = дані_обʼєкта_типу_структура;

    п = 0;
    поки п < дані_структури.дії.розмір {
      контекст.середовище = дані_обʼєкта_типу_структура.дії;

      змінна дані_дії = дані_структури.дії.дані[п].дані як розбирач::ДаніСполукиДія;

      змінна результат = скомпілювати_дію_як_елемент_тіла(
        компілятор,
        контекст,
        дані_дії,
        дані_структури.дії.дані[п].місцезнаходження
      );

      якщо результат.помилка != пусто {
        вернути РезультатОбʼєктаТипу { помилка = результат.помилка, ... };
      }

      п += 1;
    }

    вернути РезультатОбʼєктаТипу { обʼєкт_типу_структура, помилка = пусто };
  }

  дія скомпілювати_структуру_як_елемент_тіла(компілятор: *Компілятор, контекст: Контекст, дані_структури: розбирач::ДаніСполукиСтруктура, місцезнаходження: розбирач::Місцезнаходження) -> РезультатОбʼєктаТипу {
    якщо не дані_структури.ідентифікатор.заповнено {
      вернути РезультатОбʼєктаТипу {
        ...,
        помилка = створити_помилку(
          компілятор,
          "Неможливо декларувати структуру без назви.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        )
      };
    }

    змінна назва_кд = кд { пусто, 0 };

    якщо не розбирач::отримати_значення_слова(
      контекст.розібраний_текст.текст_коду.значення,
      контекст.розібраний_текст.слова.дані[дані_структури.ідентифікатор.значення.позиція_слова],
      назва_кд::адреса
    ) {
      вернути РезультатОбʼєктаТипу {
        ...,
        помилка = створити_помилку(
          компілятор,
          "Не вдалось отримати назву структури.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        )
      };
    }

    змінна назва = отримати_назву(компілятор, назва_кд);

    змінна обʼєкт_з_середовища: Обʼєкт;
    якщо знайти_в_середовищі(контекст.середовище, назва, обʼєкт_з_середовища::адреса) {
      якщо обʼєкт_з_середовища.вид == ВидОбʼєктаТип {
        змінна обʼєкт_типу = обʼєкт_з_середовища.дані як *ОбʼєктТипу;

        якщо обʼєкт_типу.вид == ВидОбʼєктаТипуСтруктура {
          змінна дані_обʼєкта_типу_структура = обʼєкт_типу.дані як ДаніОбʼєктаТипуСтруктура;

          змінна результат_доповнення = скомпілювати_структуру(
            компілятор,
            контекст,
            пусто,
            обʼєкт_типу,
            дані_структури,
            місцезнаходження
          );

          якщо результат_доповнення.помилка != пусто {
            вернути РезультатОбʼєктаТипу { пусто, помилка = результат_доповнення.помилка };
          }

          вернути РезультатОбʼєктаТипу { обʼєкт_типу, помилка = пусто };
        }
      }

      вернути РезультатОбʼєктаТипу {
        ...,
        помилка = створити_помилку_субʼєкт_вже_визначено(
          компілятор,
          Місцезнаходження { контекст.розібраний_текст, дані_структури.ідентифікатор.значення.позиція_слова },
          назва_кд
        )
      };
    }

    змінна результат = скомпілювати_структуру(
      компілятор,
      контекст,
      пусто,
      пусто,
      дані_структури,
      місцезнаходження
    );

    якщо результат.помилка != пусто {
      вернути результат;
    }

    змінити_в_середовищі(контекст.середовище, назва, Обʼєкт { ВидОбʼєктаТип, результат.обʼєкт_типу });

    вернути результат;
  }
