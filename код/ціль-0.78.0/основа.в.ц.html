структура кд {
  дані: *п8;
  розмір: природне;

  внутрішня дія рівн(я: *Я, значення: кд) -> логічне {
    змінна я_розмір = я.розмір;
    змінна я_дані = я.дані;

    якщо я_розмір != значення.розмір {
      вернути ні;
    }

    змінна п: природне = 0;
    поки п < я_розмір {
      якщо я_дані[п] != значення.дані[п] {
        вернути ні;
      }

      п += 1;
    }

    вернути так;
  }

  внутрішня дія поч_з(я: *Я, значення: кд) -> логічне {
    змінна я_розмір = я.розмір;
    змінна я_дані = я.дані;

    якщо я_розмір < значення.розмір {
      вернути ні;
    }

    змінна п: природне = 0;
    поки п < значення.розмір {
      якщо я_дані[п] != значення.дані[п] {
        вернути ні;
      }

      п += 1;
    }

    вернути так;
  }


  внутрішня дія зак_на(я: *Я, значення: кд) -> логічне {
    змінна я_розмір = я.розмір;
    змінна я_дані = я.дані;

    якщо я_розмір < значення.розмір {
      вернути ні;
    }

    змінна п: природне = 0;
    поки п < значення.розмір {
      якщо я_дані[я_розмір - значення.розмір + п] != значення.дані[п] {
        вернути ні;
      }

      п += 1;
    }

    вернути так;
  }
}
