взяти визначення КЦ;

простір КЦ {
  дія виділити_обʼєкт_дії(компілятор: *Компілятор) -> *ОбʼєктДії {
    змінна обʼєкт_дії = біб::виділити<ОбʼєктДії>(1);
    обʼєкт_дії.наступний = пусто;
    обʼєкт_дії.видимість = ВидимістьВнутрішня;
    обʼєкт_дії.назва = пусто;
    обʼєкт_дії.місцезнаходження = Місцезнаходження { пусто, 0 };
    обʼєкт_дії.місцезнаходження_назви = Місцезнаходження { пусто, 0 };
    обʼєкт_дії.повна_назва = пусто;
    обʼєкт_дії.параметри = Параметри { 0, пусто, 0 };
    обʼєкт_дії.тип_результату = компілятор.тип_ніщо;
    обʼєкт_дії.місцезнаходження_типу_результату = Місцезнаходження { пусто, 0 };
    обʼєкт_дії.тип = пусто;
    обʼєкт_дії.пм_інструкція = пусто;
    обʼєкт_дії.має_тіло = природне(ні);
    обʼєкт_дії.кроки = Кроки { 0, пусто, 0 };
    обʼєкт_дії.стрибки = Стрибки { 0, пусто, 0 };
    якщо компілятор.перший_обʼєкт_дії == пусто {
      компілятор.перший_обʼєкт_дії = обʼєкт_дії;
    }
    якщо компілятор.останній_обʼєкт_дії == пусто {
      компілятор.останній_обʼєкт_дії = обʼєкт_дії;
    } інакше {
      компілятор.останній_обʼєкт_дії.наступний = обʼєкт_дії;
      компілятор.останній_обʼєкт_дії = обʼєкт_дії;
    }
    вернути обʼєкт_дії;
  }

  дія знищити_обʼєкт_дії(компілятор: *Компілятор, обʼєкт_дії: *ОбʼєктДії) {
    знищити_назву_якщо_копія(компілятор, обʼєкт_дії.назва);
    знищити_назву_якщо_копія(компілятор, обʼєкт_дії.повна_назва);
    знищити_параметри(компілятор, обʼєкт_дії.параметри);
    знищити_кроки(компілятор, обʼєкт_дії.кроки);
    знищити_стрибки(компілятор, обʼєкт_дії.стрибки);

    біб::звільнити(обʼєкт_дії);
  }
}

простір КЦ {
  дія обʼєкт_дії_виконати(компілятор: *Компілятор, контекст: Контекст, місцезнаходження: розбирач::Місцезнаходження, обʼєкт_дії: *ОбʼєктДії, кількість_аргументів: природне, аргументи: *Аргумент, вихід: *Обʼєкт, вихід_помилки: **Помилка) -> логічне {
    змінна обʼєкт_значення: ОбʼєктЗначення;
    обʼєкт_значення.тип = обʼєкт_дії.тип;
    обʼєкт_значення.пм_значення = обʼєкт_дії.пм_інструкція як *ПМ::Значення;
    вернути обʼєкт_значення_виконати(компілятор, контекст, місцезнаходження, обʼєкт_значення, кількість_аргументів, аргументи, вихід, вихід_помилки);
  }
}
