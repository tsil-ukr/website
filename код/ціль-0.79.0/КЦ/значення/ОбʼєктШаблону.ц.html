взяти визначення КЦ;

простір КЦ {
  дія знищити_обʼєкт_шаблону(компілятор: *Компілятор, обʼєкт_шаблону: *ОбʼєктШаблону) {
    знищити_назву_якщо_копія(компілятор, обʼєкт_шаблону.назва);
    знищити_параметри(компілятор, обʼєкт_шаблону.параметри);

    біб::звільнити_реєстр<ГотовіАргументиШаблону, Обʼєкт>(обʼєкт_шаблону.готові);

    біб::звільнити(обʼєкт_шаблону);
  }
}

простір КЦ {
  дія обʼєкт_шаблону_отримати_за_позицією(
    компілятор: *Компілятор,
    контекст: Контекст,
    місцезнаходження: розбирач::Місцезнаходження,
    обʼєкт_шаблону: *ОбʼєктШаблону,
    кількість_аргументів: природне,
    аргументи: *Аргумент,
    очікуваний_тип: *ОбʼєктТипу,
    вихід: *Обʼєкт,
    вихід_помилки: **Помилка
  ) -> логічне {
    якщо кількість_аргументів != обʼєкт_шаблону.параметри.розмір {
      вихід_помилки::вміст = створити_помилку(
        компілятор,
        "Невірна кількість аргументів шаблону",
        створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
      );
      вернути ні;
    }

    змінна успіх = ні;
    змінна помилка: *Помилка;

    змінна контекст_виконання_шаблону = біб::виділити<КонтекстКомпіляціїВиконанняШаблону>(1);
    контекст_виконання_шаблону.обʼєкт_шаблону = обʼєкт_шаблону;
    контекст_виконання_шаблону.готові_аргументи_шаблону = ГотовіАргументиШаблону { кількість_аргументів, біб::виділити<ГотовийАргументШаблону>(кількість_аргументів) };
    змінна середовище_шаблону = створити_середовище(компілятор, обʼєкт_шаблону.контекст_компіляції.середовище);
    змінна п: природне = 0;
    поки п < кількість_аргументів {
      змінна аргумент = аргументи[п];

      змінна от: *ОбʼєктТипу;

      успіх = отримати_як_тип(
        компілятор,
        контекст,
        аргумент.місцезнаходження,
        аргумент.обʼєкт,
        ні,
        от::адреса,
        помилка::адреса
      );
      якщо не успіх {
        знищити_готові_аргументи_шаблону(компілятор, контекст_виконання_шаблону.готові_аргументи_шаблону);
        біб::звільнити(контекст_виконання_шаблону);
        вихід_помилки::вміст = помилка;
        вернути ні;
      }

      контекст_виконання_шаблону.готові_аргументи_шаблону.дані[п] = ГотовийАргументШаблону {
        ВидАргументаГотовихШаблонуОбʼєктТипу,
        от
      };

      змінити_в_середовищі(
        середовище_шаблону,
        обʼєкт_шаблону.параметри.дані[п].назва,
        Обʼєкт { ВидОбʼєктаТип, дані = от }
      );

      п += 1;
    }
    змінна обʼєкт_з_реєстру_готових: Обʼєкт;

    якщо біб::знайти_в_реєстрі<ГотовіАргументиШаблону, Обʼєкт>(обʼєкт_шаблону.готові, контекст_виконання_шаблону.готові_аргументи_шаблону, обʼєкт_з_реєстру_готових::адреса) {
      знищити_готові_аргументи_шаблону(компілятор, контекст_виконання_шаблону.готові_аргументи_шаблону);
      біб::звільнити(контекст_виконання_шаблону);
      вихід::вміст = обʼєкт_з_реєстру_готових;
      вернути так;
    }

    успіх = скомпілювати_значення(
      компілятор,
      Контекст {
        розібраний_текст = обʼєкт_шаблону.контекст_компіляції.розібраний_текст,
        середовище = середовище_шаблону,
        кк_дії = обʼєкт_шаблону.контекст_компіляції.кк_дії
      },
      контекст_виконання_шаблону,
      обʼєкт_шаблону.сполука,
      пусто,
      вихід,
      вихід_помилки
    );

    знищити_готові_аргументи_шаблону(компілятор, контекст_виконання_шаблону.готові_аргументи_шаблону);
    біб::звільнити(контекст_виконання_шаблону);

    вернути успіх;
  }
}

простір КЦ {
  дія обʼєкт_шаблону_виконати(компілятор: *Компілятор, контекст: Контекст, місцезнаходження: розбирач::Місцезнаходження, обʼєкт_шаблону: *ОбʼєктШаблону, кількість_аргументів: природне, аргументи: *Аргумент, вихід: *Обʼєкт, вихід_помилки: **Помилка) -> логічне {
    вихід_помилки::вміст = створити_помилку(
      компілятор,
      "Не вдалось скомпілювати",
      створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
    );
    вернути ні;
  }
}

простір КЦ {
  дія обʼєкт_шаблону_виконати_шаблон(
    компілятор: *Компілятор,
    контекст: Контекст,
    місцезнаходження: розбирач::Місцезнаходження,
    обʼєкт_шаблону: *ОбʼєктШаблону,
    кількість_аргументів: природне,
    аргументи: *Аргумент,
    вихід: *Обʼєкт,
    вихід_помилки: **Помилка
  ) -> логічне {

      компілятор,
      контекст,
      місцезнаходження,
      обʼєкт_шаблону,
      кількість_аргументів,
      аргументи,
      пусто,
      вихід,
      вихід_помилки
    );
  }
