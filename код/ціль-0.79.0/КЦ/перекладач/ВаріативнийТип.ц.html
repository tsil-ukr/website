взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_варіативний_тип(
    компілятор: *Компілятор,
    контекст: Контекст,
    дані_варіативного_типу: розбирач::ДаніСполукиВаріативнийТип,
    місцезнаходження: розбирач::Місцезнаходження,
    вихід: **ОбʼєктТипу,
    вихід_помилки: **Помилка,
  ) -> логічне {
    змінна успіх = ні;

    змінна типи = біб::виділити<*ОбʼєктТипу>(дані_варіативного_типу.типи.розмір);

    змінна п: природне = 0;

    поки п < дані_варіативного_типу.типи.розмір {
      успіх = скомпілювати_значення_як_тип(
        компілятор,
        контекст,
        дані_варіативного_типу.типи.дані[п],
        так,
        типи[п]::адреса,
        вихід_помилки
      );
      якщо не успіх {
        біб::звільнити(типи);
        вернути ні;
      }

      п += 1;
    }

    вихід::вміст = створити_варіативний_тип(
      компілятор,
      дані_варіативного_типу.типи.розмір,
      типи
    );
    вернути так;
  }
}
