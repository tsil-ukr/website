взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_змінити_елемент(
    компілятор: *Компілятор,
    контекст: Контекст,
    дані_змінити_елемент: розбирач::ДаніСполукиЗмінитиЕлемент,
    місцезнаходження: розбирач::Місцезнаходження,
    вихід: *Обʼєкт,
    вихід_помилки: **Помилка
  ) -> логічне {
    змінна успіх = ні;

    змінна обʼєкт_ліво: Обʼєкт;
    змінна обʼєкт_ключа: Обʼєкт;
    змінна обʼєкт_значення: Обʼєкт;

    успіх = скомпілювати_значення(
      компілятор,
      контекст,
      пусто,
      дані_змінити_елемент.ліво,
      пусто,
      обʼєкт_ліво::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    успіх = скомпілювати_значення(
      компілятор,
      контекст,
      пусто,
      дані_змінити_елемент.ключ,
      пусто,
      обʼєкт_ключа::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    успіх = скомпілювати_значення(
      компілятор,
      контекст,
      пусто,
      дані_змінити_елемент.значення,
      пусто,
      обʼєкт_значення::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    вернути обʼєкт_змінити_за_позицією(
      компілятор,
      контекст,
      місцезнаходження,
      обʼєкт_ліво,
      Аргумент { обʼєкт_ключа, дані_змінити_елемент.ключ.місцезнаходження },
      Аргумент { обʼєкт_значення, дані_змінити_елемент.значення.місцезнаходження },
      вихід,
      вихід_помилки
    );
  }
}
