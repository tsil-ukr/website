взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_однооперацію(
    компілятор: *Компілятор,
    контекст: Контекст,
    дані_однооперації: розбирач::ДаніСполукиОднооперація,
    місцезнаходження: розбирач::Місцезнаходження,
    вихід: *Обʼєкт,
    вихід_помилки: **Помилка
  ) -> логічне {
    змінна успіх = ні;

    змінна обʼєкт_значення: Обʼєкт;

    успіх = скомпілювати_значення(
      компілятор,
      контекст,
      пусто,
      дані_однооперації.значення,
      пусто,
      обʼєкт_значення::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    якщо дані_однооперації.вид == розбирач::ВидОдноопераціїЗаперечення {
      вернути обʼєкт_заперечити(компілятор, контекст, місцезнаходження, обʼєкт_значення, вихід, вихід_помилки);
    } інакше якщо дані_однооперації.вид == розбирач::ВидОдноопераціїПлюс {
      вихід::вміст = обʼєкт_значення;
      вернути так;
    } інакше якщо дані_однооперації.вид == розбирач::ВидОдноопераціїМінус {
      вернути обʼєкт_унарний_мінус(компілятор, контекст, місцезнаходження, обʼєкт_значення, вихід, вихід_помилки);
    } інакше якщо дані_однооперації.вид == розбирач::ВидОдноопераціїДвійковеЗаперечення {
      вернути обʼєкт_двійкове_заперечення(компілятор, контекст, місцезнаходження, обʼєкт_значення, вихід, вихід_помилки);
    } інакше якщо дані_однооперації.вид == розбирач::ВидОдноопераціїЗірочка {
      вернути обʼєкт_зірочка(компілятор, контекст, місцезнаходження, обʼєкт_значення, вихід, вихід_помилки);
    }

    вихід_помилки::вміст = створити_помилку(
      компілятор,
      "Однооперацію не втілено.",
      створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
    );
    вернути ні;
  }
}
