взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_отримати_властивість(
    компілятор: *Компілятор,
    контекст: Контекст,
    дані_отримати_властивість: розбирач::ДаніСполукиОтриматиВластивість,
    місцезнаходження: розбирач::Місцезнаходження,
    очікуваний_тип: *ОбʼєктТипу,
    вихід: *Обʼєкт,
    вихід_помилки: **Помилка
  ) -> логічне {
    змінна успіх = ні;

    змінна обʼєкт_ліво: Обʼєкт;

    успіх = скомпілювати_значення(
      компілятор,
      контекст,
      пусто,
      дані_отримати_властивість.ліво,
      пусто,
      обʼєкт_ліво::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    змінна назва_кд = кд { пусто, 0 };

    якщо не розбирач::отримати_значення_ідентифікатора(
      контекст.розібраний_текст.текст_коду.значення,
      контекст.розібраний_текст.слова,
      дані_отримати_властивість.ідентифікатор,
      назва_кд::адреса
    ) {
      вихід_помилки::вміст = створити_помилку(
        компілятор,
        "Не вдалось отримати назву властивості.",
        створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
      );
      вернути ні;
    }

    змінна назва = отримати_назву(компілятор, назва_кд);

    вернути обʼєкт_отримати_властивість(
      компілятор,
      контекст,
      місцезнаходження,
      обʼєкт_ліво,
      назва,
      очікуваний_тип,
      вихід,
      вихід_помилки
    );
  }
}
