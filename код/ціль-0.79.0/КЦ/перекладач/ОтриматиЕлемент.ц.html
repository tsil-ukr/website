взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_отримати_елемент(
    компілятор: *Компілятор,
    контекст: Контекст,
    дані_отримати_елемент: розбирач::ДаніСполукиОтриматиЕлемент,
    місцезнаходження: розбирач::Місцезнаходження,
    очікуваний_тип: *ОбʼєктТипу,
    вихід: *Обʼєкт,
    вихід_помилки: **Помилка
  ) -> логічне {
    змінна успіх = ні;

    змінна обʼєкт_ліво: Обʼєкт;

    успіх = скомпілювати_значення(
      компілятор,
      контекст,
      пусто,
      дані_отримати_елемент.ліво,
      пусто,
      обʼєкт_ліво::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    змінна аргументи = біб::виділити<Аргумент>(
      дані_отримати_елемент.аргументи.розмір
    );

    змінна п: природне = 0;
    поки п < дані_отримати_елемент.аргументи.розмір {
      змінна обʼєкт_аргумента: Обʼєкт;

      успіх = скомпілювати_значення(
        компілятор,
        контекст,
        пусто,
        дані_отримати_елемент.аргументи.дані[п].значення,
        пусто,
        обʼєкт_аргумента::адреса,
        вихід_помилки
      );
      якщо не успіх {
        біб::звільнити(аргументи);
        вернути ні;
      }

      аргументи[п] = Аргумент {
        обʼєкт = обʼєкт_аргумента,
        місцезнаходження = дані_отримати_елемент.аргументи.дані[п].значення.місцезнаходження
      };

      п += 1;
    }

    вернути обʼєкт_отримати_за_позицією(
      компілятор,
      контекст,
      місцезнаходження,
      обʼєкт_ліво,
      дані_отримати_елемент.аргументи.розмір,
      аргументи,
      очікуваний_тип,
      вихід,
      вихід_помилки
    );
  }
}
