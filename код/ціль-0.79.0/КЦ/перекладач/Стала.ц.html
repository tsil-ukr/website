взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_визначити_сталу(
    компілятор: *Компілятор,
    контекст: Контекст,
    дані_сталої: розбирач::ДаніСполукиСтала,
    місцезнаходження: розбирач::Місцезнаходження,
    вихід_помилки: **Помилка,
  ) -> логічне {
    змінна успіх = ні;

    змінна назва_кд = кд { пусто, 0 };

    якщо не розбирач::отримати_значення_слова(
      контекст.розібраний_текст.текст_коду.значення,
      контекст.розібраний_текст.слова.дані[дані_сталої.ідентифікатор.позиція_слова],
      назва_кд::адреса
    ) {
      вихід_помилки::вміст = створити_помилку(
        компілятор,
        "Не вдалось отримати назву сталої.",
        створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
      );
      вернути ні;
    }

    змінна назва = отримати_назву(компілятор, назва_кд);

    змінна обʼєкт_з_середовища: Обʼєкт;
    якщо знайти_в_середовищі(контекст.середовище, назва, обʼєкт_з_середовища::адреса) {
      вихід_помилки::вміст = створити_помилку(
        компілятор,
        "Субʼєкт вже визначено.",
        створити_місцезнаходження_помилки_з_позиції_слова(компілятор, контекст.розібраний_текст, дані_сталої.ідентифікатор.позиція_слова)
      );
      вернути ні;
    }

    змінна обʼєкт: Обʼєкт;
    змінна обʼєкт_значення: Обʼєкт;

    якщо контекст.кк_дії == пусто {

      успіх = скомпілювати_значення(
        компілятор,
        контекст,
        пусто,
        дані_сталої.значення,
        пусто,
        обʼєкт_значення::адреса,
        вихід_помилки
      );
      якщо не успіх {
        вернути ні;
      }

      якщо обʼєкт_значення.вид == ВидОбʼєктаЗначення {
        обʼєкт = обʼєкт_значення;
      } інакше якщо обʼєкт_значення.вид == ВидОбʼєктаСиреЗначення {
        обʼєкт = обʼєкт_значення;
      } інакше {
        вихід_помилки::вміст = створити_помилку(
          компілятор,
          "Неможливо створити сталу.",
          створити_місцезнаходження_помилки_з_позиції_слова(компілятор, контекст.розібраний_текст, дані_сталої.ідентифікатор.позиція_слова)
        );
        вернути ні;
      }
    } інакше {
      успіх = скомпілювати_значення(
        компілятор,
        контекст,
        пусто,
        дані_сталої.значення,
        пусто,
        обʼєкт_значення::адреса,
        вихід_помилки
      );
      якщо не успіх {
        вернути ні;
      }

      якщо обʼєкт_значення.вид == ВидОбʼєктаНезавантаженеЗначення {
        змінна обʼєкт_незавантаженого_значення = обʼєкт_значення.дані як ОбʼєктНезавантаженогоЗначення;
        змінна пм_значення = ПМ::вказівка_прочитати(
          контекст.кк_дії.поточний_крок.пм_крок,
          обʼєкт_незавантаженого_значення.тип.пм_тип,
          обʼєкт_незавантаженого_значення.пм_значення
        );
        обʼєкт = Обʼєкт {
          ВидОбʼєктаЗначення,
          ОбʼєктЗначення { обʼєкт_незавантаженого_значення.тип, пм_значення }
        };
      } інакше якщо обʼєкт_значення.вид == ВидОбʼєктаЗначення {
        обʼєкт = обʼєкт_значення;
      } інакше якщо обʼєкт_значення.вид == ВидОбʼєктаСиреЗначення {
        обʼєкт = обʼєкт_значення;
      } інакше {
        вихід_помилки::вміст = створити_помилку(
          компілятор,
          "Неможливо створити сталу.",
          створити_місцезнаходження_помилки_з_позиції_слова(компілятор, контекст.розібраний_текст, дані_сталої.ідентифікатор.позиція_слова)
        );
        вернути ні;
      }
    }

    змінити_в_середовищі(
      контекст.середовище,
      назва,
      обʼєкт
    );

    вернути так;
  }
}
