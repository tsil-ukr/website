взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_тип_дії(
    компілятор: *Компілятор,
    контекст: Контекст,
    дані_типу_дії: розбирач::ДаніСполукиТипДії,
    місцезнаходження: розбирач::Місцезнаходження,
    вихід: **ОбʼєктТипу,
    вихід_помилки: **Помилка,
  ) -> логічне {
    змінна успіх = ні;
    змінна рез_обʼєкт_типу: *ОбʼєктТипу;
    змінна параметри: Параметри;

    успіх = скомпілювати_параметри_дії(
      компілятор,
      контекст,
      пусто,
      дані_типу_дії.параметри,
      місцезнаходження,
      параметри::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    успіх = скомпілювати_значення_як_тип(
      компілятор,
      контекст,
      дані_типу_дії.тип_результату,
      так,
      рез_обʼєкт_типу::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    вихід::вміст = створити_тип_дії(
      компілятор,
      параметри,
      рез_обʼєкт_типу
    );
    вернути так;
  }
}
