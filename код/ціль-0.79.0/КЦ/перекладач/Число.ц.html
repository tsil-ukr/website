    дані_числа: розбирач::ДаніСполукиЧисло,
    місцезнаходження: розбирач::Місцезнаходження,
    очікуваний_тип: *ОбʼєктТипу,
    вихід: *Обʼєкт,
    вихід_помилки: **Помилка
  ) -> логічне {
    змінна успіх = ні;
    змінна помилка: *Помилка;

    змінна значення_кд = кд { пусто, 0 };

    якщо не розбирач::отримати_значення_слова(
      контекст.розібраний_текст.текст_коду.значення,
      контекст.розібраний_текст.слова.дані[дані_числа.позиція_слова],
      значення_кд::адреса
    ) {
      вихід_помилки::вміст = створити_помилку(
        компілятор,
        "Не вдалось скомпілювати число.",
        створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
      );
      вернути ні;
    }

    якщо очікуваний_тип == пусто {
      змінна обʼєкт_сирого_значення = виділити_обʼєкт_сирого_значення(компілятор);
      обʼєкт_сирого_значення.дані = виділити_копію_кд(компілятор, значення_кд);

      вихід::вміст = Обʼєкт {
        ВидОбʼєктаСиреЗначення,
        обʼєкт_сирого_значення
      };
      вернути так;
    } інакше {
      змінна обʼєкт_значення_числа: ОбʼєктЗначення;

      успіх = розібрати_число(
        компілятор,
        контекст,
        місцезнаходження,
        значення_кд,
        очікуваний_тип,
        обʼєкт_значення_числа::адреса,
        помилка::адреса
      );
      якщо не успіх {
        вихід_помилки::вміст = помилка;
        вернути ні;
      }

      вихід::вміст = Обʼєкт {
        ВидОбʼєктаЗначення,
        обʼєкт_значення_числа
      };
      вернути так;
    }

    вихід_помилки::вміст = створити_помилку(
      компілятор,
      "Не вдалось скомпілювати число.",
      створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
    );
    вернути ні;
  }
}
