зовнішня дія __КЦ__кланг_лінк(
  вих: *біб::Шлях,
  кф: природне,
  сф: *біб::Шлях,
  кллвір: природне,
  сллвмір: *біб::Шлях
) -> ц32;

внутрішня дія скомпілювати_біб_ц_якщо_треба(
  шлях_до_основи: біб::Шлях,
  шлях_до_біб_ц: біб::Шлях,
  вихід_ллвмір: *біб::Шлях,
) -> ц32 {
  змінна шлях_до_біб_ц_ллвмір: біб::Шлях;
  якщо біб::клонувати_шлях(шлях_до_біб_ц, шлях_до_біб_ц_ллвмір::адреса) != 0 {
    біб::надрукувати("не біб::клонувати_шлях(шлях_до_біб_ц, шлях_до_біб_ц_ллвмір::адреса)");
    вернути 1;
  }

  біб::доповнити_шлях_текстом(шлях_до_біб_ц_ллвмір::адреса, ".ллвмір");

  змінна кр = ціль_скомпілювати(
    шлях_до_основи,
    цілеспрямовано::Опції<ю8> { 0, пусто },
    цілеспрямовано::Опції<ю8> { 0, пусто },
    ю8 { шлях_до_біб_ц.дані, шлях_до_біб_ц.розмір },
    цілеспрямовано::Опції<ю8> { 0, пусто },
    ю8 { шлях_до_біб_ц_ллвмір.дані, шлях_до_біб_ц_ллвмір.розмір },
    цілеспрямовано::Опції<ю8> { 0, пусто },
    так,
  );

  якщо кр == 0 {
    вихід_ллвмір::вміст = шлях_до_біб_ц_ллвмір;
  } інакше {
    біб::звільнити(шлях_до_біб_ц_ллвмір.дані);
  }

  вернути кр;
}

дія ціль_зібрати(
  шлях_до_основи: біб::Шлях,
  шлях_до_мейн: біб::Шлях,
  шлях_до_біб: біб::Шлях,
  шлях_до_біб_ц: біб::Шлях,
  шлях_до_біб_сі: біб::Шлях,
  опції_програми: цілеспрямовано::Опції<ю8>,
  опції_команди: цілеспрямовано::Опції<ю8>,
  входи: цілеспрямовано::ВходиКоманди<ю8>,
  вихід: ю8,
  опції_виходу: цілеспрямовано::Опції<ю8>,
) -> ц32 {
  змінна кр = ц32(0);
  змінна п = природне(0);
  змінна вх_файли = біб::зробити_накопичувач<біб::Шлях>(0);
  змінна ллвмір_файли = біб::зробити_накопичувач<біб::Шлях>(0);
  змінна шлях_до_біб_ц_ллвмір: біб::Шлях;

  кр = скомпілювати_біб_ц_якщо_треба(
    шлях_до_основи,
    шлях_до_біб_ц,
    шлях_до_біб_ц_ллвмір::адреса,
  );

  якщо кр != 0 {
    вернути кр;
  }

  біб::покласти_в_накопичувач<біб::Шлях>(вх_файли::адреса, шлях_до_мейн);
  біб::покласти_в_накопичувач<біб::Шлях>(вх_файли::адреса, шлях_до_біб_сі);

  біб::покласти_в_накопичувач<біб::Шлях>(ллвмір_файли::адреса, шлях_до_біб_ц_ллвмір);

  поки п < входи.розмір {
    змінна вхід = входи.дані[п];

    якщо вхід.значення.зак_на(ю8".ц") {
      змінна вихроз = ю8".ллвмір";
      змінна компвихід = ю8 {
        біб::виділити<п8>(вхід.значення.розмір + вихроз.розмір),
        вхід.значення.розмір + вихроз.розмір,
      };

      змінна пп = природне(0);

      поки пп < вхід.значення.розмір {
        компвихід.дані[пп] = вхід.значення.дані[пп];

        пп += 1;
      }

      пп = 0;
      поки пп < вихроз.розмір {
        компвихід.дані[вхід.значення.розмір + пп] = вихроз.дані[пп];

        пп += 1;
      }

      // біб::вивести("ціль ");
      // біб::вивести(компвихід);
      // біб::вивести(" скомпілювати ");
      // біб::надрукувати(вхід.значення);

      якщо не цілеспрямовано::має_опцію<ю8>(вхід.опції, ю8"біб", ю8"Б") {
        цілеспрямовано::додати_опцію<ю8>(
          вхід.опції::адреса,
          ю8"біб",
          ні,
          ю8 { шлях_до_біб.дані, шлях_до_біб.розмір }
        );
      }

      кр = ціль_скомпілювати(
        шлях_до_основи,
        опції_програми,
        цілеспрямовано::Опції<ю8> { 0, пусто },
        вхід.значення,
        вхід.опції,
        компвихід,
        цілеспрямовано::Опції<ю8> { 0, пусто },
        так,
      );

      якщо кр != 0 {
        змінна пп = природне(0);
        поки пп < ллвмір_файли.розмір {
          біб::видалити_за_шляхом(ллвмір_файли.дані[пп]);
          пп += 1;
        }
        вернути кр;
      }

      змінна вихшлях: біб::Шлях;

      якщо біб::шлях_з_ю8(компвихід, вихшлях::адреса) != 0 {
        біб::надрукувати("біб::шлях_з_ю8(компвихід, вихшлях::адреса) != 0");
        змінна пп = природне(0);
        поки пп < ллвмір_файли.розмір {
          біб::видалити_за_шляхом(ллвмір_файли.дані[пп]);
          пп += 1;
        }
        вернути 1;
      }

      біб::покласти_в_накопичувач<біб::Шлях>(ллвмір_файли::адреса, вихшлях);

      біб::звільнити(компвихід.дані);
    } інакше {
      змінна ш: біб::Шлях;
      якщо біб::шлях_з_ю8(вхід.значення, ш::адреса) != 0 {
        біб::надрукувати("біб::шлях_з_ю8(вхід.значення, ш::адреса) != 0");
        змінна пп = природне(0);
        поки пп < ллвмір_файли.розмір {
          біб::видалити_за_шляхом(ллвмір_файли.дані[пп]);
          пп += 1;
        }
        вернути 1;
      }
      біб::покласти_в_накопичувач<біб::Шлях>(вх_файли::адреса, ш);
    }

    п += 1;
  }

  змінна вихшлях: біб::Шлях;

  якщо біб::шлях_з_ю8(вихід, вихшлях::адреса) != 0 {
    біб::надрукувати("біб::шлях_з_ю8(вихід, вихшлях::адреса) != 0");
    змінна пп = природне(0);
    поки пп < ллвмір_файли.розмір {
      біб::видалити_за_шляхом(ллвмір_файли.дані[пп]);
      пп += 1;
    }
    вернути 1;
  }

  кр = __КЦ__кланг_лінк(
    вихшлях::адреса,
    вх_файли.розмір,
    вх_файли.дані,
    ллвмір_файли.розмір,
    ллвмір_файли.дані
  );

  якщо кр == 0 {
    змінна пп = природне(0);
    поки пп < ллвмір_файли.розмір {
      біб::видалити_за_шляхом(ллвмір_файли.дані[пп]);
      пп += 1;
    }
  }

  вернути кр;
}
