дія зробити_шлях_до_основи(шлях_до_цілі: ю8, вихід: *ю8) -> логічне {
  змінна шлях_до_папки_цілі = ю8 { шлях_до_цілі.дані, шлях_до_цілі.розмір };

  змінна глиб = природне(0);

  поки шлях_до_папки_цілі.розмір > 0 {
    якщо шлях_до_папки_цілі.дані[шлях_до_папки_цілі.розмір - 1] == 47 { // '/'
      якщо глиб == 1 {
        стрибнути далі;
      } інакше {
        глиб += 1;
      }
    }

    шлях_до_папки_цілі.розмір -= 1;
  }

  вернути ні;

далі:
  змінна назва_основи = ю8"основа.в.ц";
  шлях_до_папки_цілі.дані = біб::виділити<п8>(шлях_до_папки_цілі.розмір + назва_основи.розмір);
  змінна п: природне = 0;
  поки п < шлях_до_папки_цілі.розмір {
    шлях_до_папки_цілі.дані[п] = шлях_до_цілі.дані[п];
    п += 1;
  }
  п = 0;
  поки п < назва_основи.розмір {
    шлях_до_папки_цілі.дані[шлях_до_папки_цілі.розмір + п] = назва_основи.дані[п];
    п += 1;
  }
  шлях_до_папки_цілі.розмір += назва_основи.розмір;

  вихід::вміст = шлях_до_папки_цілі;

  вернути так;
}

дія почати(кількість_аргументів: природне, аргументи: *ю8) -> ц32 {
  змінна статус_виходу: ц32 = 0;
  змінна шлях_до_цілі: ю8;
  змінна шлях_до_основи: ю8;
  змінна шлях_до_мейн: ю8;
  змінна шлях_до_біб: ю8;
  змінна шлях_до_біб_ц: ю8;
  змінна шлях_до_біб_сі: ю8;

  __КЦ__отримати_шлях_до_цілі_як_ю8(шлях_до_цілі::адреса);

  якщо не зробити_шлях_до_основи(шлях_до_цілі, шлях_до_основи::адреса) {
    біб::надрукувати_ю8(шлях_до_цілі);
    біб::надрукувати("Не вдалось визначити шлях до основи.в.ц!");
    вернути 1;
  }

  якщо не зробити_шлях_до_мейн(шлях_до_цілі, шлях_до_мейн::адреса) {
    біб::надрукувати_ю8(шлях_до_цілі);
    біб::надрукувати("Не вдалось визначити шлях до мейн!");
    вернути 1;
  }

  якщо не зробити_шлях_до_біб(шлях_до_цілі, шлях_до_біб::адреса) {
    біб::надрукувати_ю8(шлях_до_цілі);
    біб::надрукувати("Не вдалось визначити шлях до біб!");
    вернути 1;
  }

  якщо не зробити_шлях_до_біб_ц(шлях_до_цілі, шлях_до_біб_ц::адреса) {
    біб::надрукувати_ю8(шлях_до_цілі);
    біб::надрукувати("Не вдалось визначити шлях до біб.ц!");
    вернути 1;
  }

  якщо не зробити_шлях_до_біб_сі(шлях_до_цілі, шлях_до_біб_сі::адреса) {
    біб::надрукувати_ю8(шлях_до_цілі);
    біб::надрукувати("Не вдалось визначити шлях до біб сі!");
    вернути 1;
  }

  змінна кількість_команд: природне = 9;

  змінна доступні_команди = цілеспрямовано::ДоступніКомандиДляРозбору<ю8> { кількість_команд, біб::виділити<ю8>(кількість_команд) };
  доступні_команди.дані[0] = ю8"допомога";
  доступні_команди.дані[1] = ю8"версія";
  доступні_команди.дані[2] = ю8"скомпілювати";
  доступні_команди.дані[3] = ю8"перетворити";
  доступні_команди.дані[4] = ю8"ювкд";
  доступні_команди.дані[5] = ю8"кдвю";
  доступні_команди.дані[6] = ю8"платформи";
  доступні_команди.дані[7] = ю8"зібрати";
  доступні_команди.дані[8] = ю8"біб";

  змінна розібрана_команда = цілеспрямовано::зробити_розібрану_команду<ю8>();

  змінна успіх_розбору_команди = цілеспрямовано::розібрати_команду<ю8>(
    цілеспрямовано::АргументиДляРозбору<ю8> { кількість_аргументів - 1, аргументи[1]::адреса },
    доступні_команди,
    ю8"--",
    ю8"-",
    ю8"=",
    розібрана_команда::адреса
  );

  якщо не успіх_розбору_команди {
    біб::надрукувати("Не вдалось розібрати команду!");
    вернути 1;
  }

  якщо розібрана_команда.команда.рівн(ю8"") {
    статус_виходу = ціль_допомога(
      цілеспрямовано::Опції<ю8> { 0, пусто },
      цілеспрямовано::Опції<ю8> { 0, пусто }
    );

    стрибнути вихід;
  }

  якщо розібрана_команда.команда.рівн(ю8"допомога") {
    статус_виходу = ціль_допомога(
      розібрана_команда.опції_програми,
      розібрана_команда.опції_команди
    );

    стрибнути вихід;
  }

  якщо розібрана_команда.команда.рівн(ю8"версія") {
    статус_виходу = ціль_версія(
      розібрана_команда.опції_програми,
      розібрана_команда.опції_команди
    );

    стрибнути вихід;
  }

  якщо розібрана_команда.команда.рівн(ю8"скомпілювати") або розібрана_команда.команда.рівн(ю8"перетворити") {
    якщо розібрана_команда.входи.розмір < 1 {
      біб::надрукувати("Недостатньо вхідних!");
      статус_виходу = 1;
      стрибнути вихід;
    }

    якщо розібрана_команда.виходи.розмір == 0 {
      змінна п: природне = 0;
      поки п < розібрана_команда.входи.розмір {
        змінна вхід = розібрана_команда.входи.дані[п];

        статус_виходу = ціль_скомпілювати(
          біб::Шлях { шлях_до_основи.розмір, шлях_до_основи.дані, шлях_до_основи.розмір },
          розібрана_команда.опції_програми,
          розібрана_команда.опції_команди,
          вхід.значення,
          вхід.опції,
          ю8 { пусто, 0 },
          цілеспрямовано::Опції<ю8> { 0, пусто },
          розібрана_команда.входи.розмір == 1
        );

        якщо статус_виходу != 0 {
          стрибнути вихід;
        }

        п += 1;
      }

      стрибнути вихід;
    } інакше {
      якщо розібрана_команда.входи.розмір > 1 {
        біб::надрукувати("Команда скомпілювати з вказаними виходами може мати лише один вхід!");
        статус_виходу = 1;
        стрибнути вихід;
      }

      змінна вхід = розібрана_команда.входи.дані[0];

      змінна п: природне = 0;
      поки п < розібрана_команда.виходи.розмір {
        змінна вихід = розібрана_команда.виходи.дані[п];

        статус_виходу = ціль_скомпілювати(
          біб::Шлях { шлях_до_основи.розмір, шлях_до_основи.дані, шлях_до_основи.розмір },
          розібрана_команда.опції_програми,
          розібрана_команда.опції_команди,
          вхід.значення,
          вхід.опції,
          вихід.значення,
          вихід.опції,
          розібрана_команда.виходи.розмір == 1
        );

        якщо статус_виходу != 0 {
          стрибнути вихід;
        }

        п += 1;
      }

      стрибнути вихід;
    }
  }

  якщо розібрана_команда.команда.рівн(ю8"зібрати") {
    якщо розібрана_команда.входи.розмір < 1 {
      біб::надрукувати("Недостатньо вхідних!");
      статус_виходу = 1;
      стрибнути вихід;
    }

    якщо розібрана_команда.виходи.розмір == 0 {
      статус_виходу = ціль_зібрати(
        біб::Шлях { шлях_до_основи.розмір, шлях_до_основи.дані, шлях_до_основи.розмір },
        біб::Шлях { шлях_до_мейн.розмір, шлях_до_мейн.дані, шлях_до_мейн.розмір },
        біб::Шлях { шлях_до_біб.розмір, шлях_до_біб.дані, шлях_до_біб.розмір },
        біб::Шлях { шлях_до_біб_ц.розмір, шлях_до_біб_ц.дані, шлях_до_біб_ц.розмір },
        біб::Шлях { шлях_до_біб_сі.розмір, шлях_до_біб_сі.дані, шлях_до_біб_сі.розмір },
        розібрана_команда.опції_програми,
        розібрана_команда.опції_команди,
        розібрана_команда.входи,
        ю8 { пусто, 0 },
        цілеспрямовано::Опції<ю8> { 0, пусто },
      );

      стрибнути вихід;
    } інакше якщо розібрана_команда.виходи.розмір == 1 {
      статус_виходу = ціль_зібрати(
        біб::Шлях { шлях_до_основи.розмір, шлях_до_основи.дані, шлях_до_основи.розмір },
        біб::Шлях { шлях_до_мейн.розмір, шлях_до_мейн.дані, шлях_до_мейн.розмір },
