простір КЦ {
  дія виділити_обʼєкт_секції(компілятор: *Компілятор) -> *ОбʼєктСекції {
    змінна обʼєкт_секції = біб::виділити<ОбʼєктСекції>(1);
    обʼєкт_секції.наступний = пусто;
    обʼєкт_секції.назва = пусто;
    обʼєкт_секції.середовище = пусто;
    якщо компілятор.перший_обʼєкт_секції == пусто {
      компілятор.перший_обʼєкт_секції = обʼєкт_секції;
    }
    якщо компілятор.останній_обʼєкт_секції == пусто {
      компілятор.останній_обʼєкт_секції = обʼєкт_секції;
    } інакше {
      компілятор.останній_обʼєкт_секції.наступний = обʼєкт_секції;
      компілятор.останній_обʼєкт_секції = обʼєкт_секції;
    }
    вернути обʼєкт_секції;
  }

  дія знищити_обʼєкт_секції(компілятор: *Компілятор, обʼєкт_секції: *ОбʼєктСекції) {
    знищити_назву_якщо_копія(компілятор, обʼєкт_секції.назва);

    біб::звільнити(обʼєкт_секції);
  }
}

простір КЦ {
  дія обʼєкт_секції_отримати_властивість_секції(
    компілятор: *Компілятор,
    контекст: Контекст,
    місцезнаходження: розбирач::Місцезнаходження,
    обʼєкт_секції: *ОбʼєктСекції,
    назва: *Назва,
    очікуваний_тип: *ОбʼєктТипу,
    вихід: *Обʼєкт,
    вихід_помилки: **Помилка
  ) -> логічне {
    змінна обʼєкт: Обʼєкт;

    якщо знайти_в_середовищі(обʼєкт_секції.середовище, назва, обʼєкт::адреса) {
      вихід::вміст = обʼєкт;
      вернути так;
    }

    вихід_помилки::вміст = створити_помилку(
      компілятор,
      "Не вдалось скомпілювати",
      створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
    );
    вернути ні;
  }
}
