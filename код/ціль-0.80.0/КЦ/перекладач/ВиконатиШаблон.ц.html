взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_виконати_шаблон(
    компілятор: *Компілятор,
    контекст: Контекст,
    дані_виконати_шаблон: розбирач::ДаніСполукиВиконатиШаблон,
    місцезнаходження: розбирач::Місцезнаходження,
    очікуваний_тип: *ОбʼєктТипу,
    вихід: *Обʼєкт,
    вихід_помилки: **Помилка
  ) -> логічне {
    змінна успіх = ні;

    змінна обʼєкт_ліво: Обʼєкт;

    успіх = скомпілювати_значення(
      компілятор,
      контекст,
      пусто,
      дані_виконати_шаблон.ліво,
      пусто,
      обʼєкт_ліво::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    змінна аргументи = біб::виділити<Аргумент>(
      дані_виконати_шаблон.аргументи.розмір
    );

    змінна п: природне = 0;
    поки п < дані_виконати_шаблон.аргументи.розмір {
      змінна обʼєкт_аргумента: Обʼєкт;

      успіх = скомпілювати_значення(
        компілятор,
        контекст,
        пусто,
        дані_виконати_шаблон.аргументи.дані[п].значення,
        пусто,
        обʼєкт_аргумента::адреса,
        вихід_помилки
      );
      якщо не успіх {
        біб::звільнити(аргументи);
        вернути ні;
      }

      аргументи[п] = Аргумент {
        обʼєкт = обʼєкт_аргумента,
        місцезнаходження = дані_виконати_шаблон.аргументи.дані[п].значення.місцезнаходження
      };

      п += 1;
    }

    успіх = обʼєкт_виконати_шаблон(
      компілятор,
      контекст,
      місцезнаходження,
      обʼєкт_ліво,
      дані_виконати_шаблон.аргументи.розмір,
      аргументи,
      вихід,
      вихід_помилки
    );

    біб::звільнити(аргументи);

    вернути успіх;
  }
}
