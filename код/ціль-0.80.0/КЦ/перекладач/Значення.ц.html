взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_значення(
    компілятор: *Компілятор,
    контекст: Контекст,
    контекст_виконання_шаблону: *КонтекстКомпіляціїВиконанняШаблону,
    сполука: *розбирач::Сполука,
    очікуваний_тип: *ОбʼєктТипу,
    вихід: *Обʼєкт,
    вихід_помилки: **Помилка
  ) -> логічне {
    змінна успіх = ні;

    змінна рез_обʼєкт_типу: *ОбʼєктТипу;
    змінна рез_обʼєкт_шаблону: *ОбʼєктШаблону;
    змінна рез_обʼєкт_дії: *ОбʼєктДії;

    якщо сполука.вид == розбирач::ВидСполукиПрочитати {
      змінна дані = сполука.дані як розбирач::ДаніСполукиПрочитати;
      вернути скомпілювати_отримати_з_середовища(компілятор, контекст, дані, сполука.місцезнаходження, очікуваний_тип, вихід, вихід_помилки);
    } інакше якщо сполука.вид == розбирач::ВидСполукиЧисло {
      змінна дані = сполука.дані як розбирач::ДаніСполукиЧисло;
      вернути скомпілювати_число(компілятор, контекст, дані, сполука.місцезнаходження, очікуваний_тип, вихід, вихід_помилки);
    } інакше якщо сполука.вид == розбирач::ВидСполукиТекст {
      змінна дані = сполука.дані як розбирач::ДаніСполукиТекст;
      вернути скомпілювати_текст(компілятор, контекст, дані, сполука.місцезнаходження, вихід, вихід_помилки);
    } інакше якщо сполука.вид == розбирач::ВидСполукиВиконатиШаблон {
      змінна дані = сполука.дані як розбирач::ДаніСполукиВиконатиШаблон;
      вернути скомпілювати_виконати_шаблон(компілятор, контекст, дані, сполука.місцезнаходження, очікуваний_тип, вихід, вихід_помилки);
    } інакше якщо сполука.вид == розбирач::ВидСполукиОперація {
      змінна дані = сполука.дані як розбирач::ДаніСполукиОперація;
      вернути скомпілювати_операцію(компілятор, контекст, дані, сполука.місцезнаходження, вихід, вихід_помилки);
    } інакше якщо сполука.вид == розбирач::ВидСполукиОднооперація {
      змінна дані = сполука.дані як розбирач::ДаніСполукиОднооперація;
      вернути скомпілювати_однооперацію(компілятор, контекст, дані, сполука.місцезнаходження, вихід, вихід_помилки);
    } інакше якщо сполука.вид == розбирач::ВидСполукиОтриматиВластивість {
      змінна дані = сполука.дані як розбирач::ДаніСполукиОтриматиВластивість;
      вернути скомпілювати_отримати_властивість(компілятор, контекст, дані, сполука.місцезнаходження, очікуваний_тип, вихід, вихід_помилки);
    } інакше якщо сполука.вид == розбирач::ВидСполукиОтриматиЕлемент {
      змінна дані = сполука.дані як розбирач::ДаніСполукиОтриматиЕлемент;
      вернути скомпілювати_отримати_елемент(компілятор, контекст, дані, сполука.місцезнаходження, очікуваний_тип, вихід, вихід_помилки);
    } інакше якщо сполука.вид == розбирач::ВидСполукиОтриматиСекційнуВластивість {
      змінна дані = сполука.дані як розбирач::ДаніСполукиОтриматиСекційнуВластивість;
      вернути скомпілювати_отримати_секційну_властивість(компілятор, контекст, дані, сполука.місцезнаходження, очікуваний_тип, вихід, вихід_помилки);
    } інакше якщо сполука.вид == розбирач::ВидСполукиОбʼєкт {
      змінна дані = сполука.дані як розбирач::ДаніСполукиОбʼєкт;
      вернути скомпілювати_обʼєкт(компілятор, контекст, дані, сполука.місцезнаходження, очікуваний_тип, вихід, вихід_помилки);
    } інакше якщо сполука.вид == розбирач::ВидСполукиВиконатиДію {
      змінна дані = сполука.дані як розбирач::ДаніСполукиВиконатиДію;
      вернути скомпілювати_виконати_дію(компілятор, контекст, дані, сполука.місцезнаходження, очікуваний_тип, вихід, вихід_помилки);
    } інакше якщо сполука.вид == розбирач::ВидСполукиДія {
      змінна дані = сполука.дані як розбирач::ДаніСполукиДія;

      успіх = скомпілювати_дію(
        компілятор,
        контекст,
        контекст_виконання_шаблону,
        пусто,
        дані,
        сполука.місцезнаходження,
        рез_обʼєкт_дії::адреса,
        вихід_помилки
      );
      якщо не успіх {
        вернути ні;
      }

      вихід::вміст = Обʼєкт { ВидОбʼєктаДія, рез_обʼєкт_дії };
      вернути так;
    } інакше якщо сполука.вид == розбирач::ВидСполукиСтруктура {
      змінна дані = сполука.дані як розбирач::ДаніСполукиСтруктура;

      успіх = скомпілювати_структуру(
        компілятор,
        контекст,
        контекст_виконання_шаблону,
        пусто,
        дані,
        сполука.місцезнаходження,
        рез_обʼєкт_типу::адреса,
        вихід_помилки
      );
      якщо не успіх {
        вернути ні;
      }

      вихід::вміст = Обʼєкт { ВидОбʼєктаТип, рез_обʼєкт_типу };
      вернути так;
    } інакше якщо сполука.вид == розбирач::ВидСполукиТипДії {
      змінна дані = сполука.дані як розбирач::ДаніСполукиТипДії;

      успіх = скомпілювати_тип_дії(
        компілятор,
        контекст,
        дані,
        сполука.місцезнаходження,
        рез_обʼєкт_типу::адреса,
        вихід_помилки
      );
      якщо не успіх {
        вернути ні;
      }

      вихід::вміст = Обʼєкт { ВидОбʼєктаТип, рез_обʼєкт_типу };
      вернути так;
    } інакше якщо сполука.вид == розбирач::ВидСполукиВаріативнийТип {
      змінна дані = сполука.дані як розбирач::ДаніСполукиВаріативнийТип;

      успіх = скомпілювати_варіативний_тип(
        компілятор,
        контекст,
        дані,
        сполука.місцезнаходження,
        рез_обʼєкт_типу::адреса,
        вихід_помилки
      );
      якщо не успіх {
        вернути ні;
      }

      вихід::вміст = Обʼєкт { ВидОбʼєктаТип, рез_обʼєкт_типу };
      вернути так;
    } інакше якщо сполука.вид == розбирач::ВидСполукиШаблон {
      змінна дані = сполука.дані як розбирач::ДаніСполукиШаблон;

      успіх = скомпілювати_шаблон(
        компілятор,
        контекст,
        дані,
        сполука.місцезнаходження,
        рез_обʼєкт_шаблону::адреса,
        вихід_помилки
      );
      якщо не успіх {
        вернути ні;
      }

      вихід::вміст = Обʼєкт { ВидОбʼєктаШаблон, рез_обʼєкт_шаблону };
      вернути так;
    }

    вихід_помилки::вміст = створити_помилку(
      компілятор,
      "Не вдалось скомпілювати значення.",
      створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, сполука.місцезнаходження)
    );
    вернути ні;
  }

  дія скомпілювати_значення_як_значення(
    компілятор: *Компілятор,
    контекст: Контекст,
    сполука: *розбирач::Сполука,
    очікуваний_тип: *ОбʼєктТипу,
    вихід: *ОбʼєктЗначення,
    вихід_помилки: **Помилка
  ) -> логічне {
    змінна успіх = ні;

    змінна обʼєкт_значення: Обʼєкт;

    успіх = скомпілювати_значення(
      компілятор,
      контекст,
      пусто,
      сполука,
      очікуваний_тип,
      обʼєкт_значення::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    вернути отримати_як_значення(
      компілятор,
      контекст,
      сполука.місцезнаходження,
      обʼєкт_значення,
      очікуваний_тип,
      вихід,
      вихід_помилки
    );
  }

  дія отримати_як_тип(
    компілятор: *Компілятор,
    контекст: Контекст,
    місцезнаходження: розбирач::Місцезнаходження,
    обʼєкт: Обʼєкт,
    перевірка_на_повність: логічне,
    вихід: **ОбʼєктТипу,
    вихід_помилки: **Помилка,
  ) -> логічне {
    якщо обʼєкт.вид != ВидОбʼєктаТип {
      вихід_помилки::вміст = створити_помилку(
        компілятор,
        "Значення повинно бути типом.",
        створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
      );
      вернути ні;
    }

    змінна обʼєкт_типу = обʼєкт.дані як *ОбʼєктТипу;

    якщо перевірка_на_повність {
      якщо перевірити_чи_тип_повний(компілятор, обʼєкт_типу) == ні {
        вихід_помилки::вміст = створити_помилку(
          компілятор,
          "Неповний тип.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        );
        вернути ні;
      }
    }

    вихід::вміст = обʼєкт_типу;
    вернути так;
  }

  дія створити_пустий_обʼєкт(
    компілятор: *Компілятор,
    контекст: Контекст,
    місцезнаходження: розбирач::Місцезнаходження,
    тип: *ОбʼєктТипу,
    вихід: *ОбʼєктЗначення,
    вихід_помилки: **Помилка
  ) -> логічне {
    якщо тип == пусто {
      вихід::вміст = компілятор.обʼєкт_значення_пусто;
      вернути так;
    }

    якщо тип.вид == ВидОбʼєктаТипуСтруктура {
      змінна пм_значення = ПМ::створити_пусте_значення(компілятор.пм_модуль, тип.пм_тип);

      вихід::вміст = ОбʼєктЗначення {
        тип = тип,
        пм_значення = пм_значення
      };
      вернути так;
    }
    якщо тип.вид == ВидОбʼєктаТипуАдреса {
      змінна пм_значення = ПМ::створити_пусте_значення(компілятор.пм_модуль, тип.пм_тип);

      вихід::вміст = ОбʼєктЗначення {
        тип = тип,
        пм_значення = пм_значення
      };
      вернути так;
    }
    якщо тип.вид == ВидОбʼєктаТипуДія {
      змінна пм_значення = ПМ::створити_пусте_значення(компілятор.пм_модуль, тип.пм_тип);

      вихід::вміст = ОбʼєктЗначення {
        тип = тип,
        пм_значення = пм_значення
      };
      вернути так;
    }

    вихід_помилки::вміст = створити_помилку(
      компілятор,
      "Неможливо зробити пустий обʼєкт.",
      створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
    );
    вернути ні;
  }

  дія отримати_як_значення(
    компілятор: *Компілятор,
    контекст: Контекст,
    місцезнаходження: розбирач::Місцезнаходження,
    обʼєкт: Обʼєкт,
    очікуваний_тип: *ОбʼєктТипу,
    вихід: *ОбʼєктЗначення,
    вихід_помилки: **Помилка
  ) -> логічне {
    якщо обʼєкт.вид == ВидОбʼєктаЗначення {
      змінна обʼєкт_значення = обʼєкт.дані як ОбʼєктЗначення;
      вихід::вміст = обʼєкт_значення;
      вернути так;
    }

    якщо обʼєкт.вид == ВидОбʼєктаНезавантаженеЗначення {
      змінна обʼєкт_незавантаженого_значення = обʼєкт.дані як ОбʼєктНезавантаженогоЗначення;

      якщо контекст.кк_дії == пусто {
        вихід_помилки::вміст = створити_помилку(
          компілятор,
          "Неможливо прочитати значення поза тілом дії.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        );
        вернути ні;
      }

      змінна пм_завантажене_значення = ПМ::вказівка_прочитати(
        контекст.кк_дії.поточний_крок.пм_крок,
        обʼєкт_незавантаженого_значення.тип.пм_тип,
        обʼєкт_незавантаженого_значення.пм_значення
      );

      змінна обʼєкт_значення = ОбʼєктЗначення {
        обʼєкт_незавантаженого_значення.тип,
        пм_завантажене_значення
      };

      вихід::вміст = обʼєкт_значення;
      вернути так;
    }

    якщо обʼєкт.вид == ВидОбʼєктаДія {
      змінна обʼєкт_дії = обʼєкт.дані як *ОбʼєктДії;

      вихід::вміст = ОбʼєктЗначення { обʼєкт_дії.тип, обʼєкт_дії.пм_інструкція як *ПМ::Значення };
      вернути так;
    }

    якщо обʼєкт.вид == ВидОбʼєктаСиреЗначення {
      змінна обʼєкт_сирого_значення = обʼєкт.дані як *ОбʼєктСирогоЗначення;

      якщо обʼєкт_сирого_значення.вид == ВидСирогоОбʼєктаЧисло {
        вернути розібрати_число(
          компілятор,
          контекст,
          місцезнаходження,
          обʼєкт_сирого_значення.дані,
          очікуваний_тип,
          вихід,
          вихід_помилки
        );
      }

      якщо обʼєкт_сирого_значення.вид == ВидСирогоОбʼєктаПусто {
        вернути створити_пустий_обʼєкт(
          компілятор,
          контекст,
          місцезнаходження,
          очікуваний_тип,
          вихід,
          вихід_помилки
        );
      }
    }

    вихід_помилки::вміст = створити_помилку(
      компілятор,
      "Значення повинно бути значенням.",
      створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
    );
    вернути ні;
  }

  дія скомпілювати_значення_як_тип(
    компілятор: *Компілятор,
    контекст: Контекст,
    сполука: *розбирач::Сполука,
    перевірка_на_повність: логічне,
    вихід: **ОбʼєктТипу,
    вихід_помилки: **Помилка,
  ) -> логічне {
    змінна успіх = ні;

    змінна обʼєкт_значення: Обʼєкт;

    успіх = скомпілювати_значення(
      компілятор,
      контекст,
      пусто,
      сполука,
      пусто,
      обʼєкт_значення::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    вернути отримати_як_тип(
      компілятор,
      контекст,
      сполука.місцезнаходження,
      обʼєкт_значення,
      перевірка_на_повність,
      вихід,
      вихід_помилки
    );
  }
