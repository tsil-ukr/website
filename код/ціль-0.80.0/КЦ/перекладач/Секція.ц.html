взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_секцію(
    компілятор: *Компілятор,
    контекст: Контекст,
    контекст_виконання_шаблону: *КонтекстКомпіляціїВиконанняШаблону,
    дані_секції: розбирач::ДаніСполукиСекція,
    місцезнаходження: розбирач::Місцезнаходження,
    вихід: **ОбʼєктСекції,
    вихід_помилки: **Помилка,
  ) -> логічне {
    змінна успіх = ні;

    змінна назва_кд = кд { пусто, 0 };

    якщо не розбирач::отримати_значення_слова(
      контекст.розібраний_текст.текст_коду.значення,
      контекст.розібраний_текст.слова.дані[дані_секції.ідентифікатор.значення.позиція_слова],
      назва_кд::адреса
    ) {
      вихід_помилки::вміст = створити_помилку(
        компілятор,
        "Не вдалось отримати назву секції.",
        створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
      );
      вернути ні;
    }

    змінна назва = отримати_назву(компілятор, назва_кд);

    змінна обʼєкт_секції = виділити_обʼєкт_секції(компілятор);

    обʼєкт_секції.назва = назва;
    обʼєкт_секції.середовище = створити_середовище_секції(компілятор, контекст.середовище, обʼєкт_секції);

    успіх = скомпілювати_тіло(
      компілятор,
      Контекст {
        розібраний_текст = контекст.розібраний_текст,
        середовище = обʼєкт_секції.середовище,
        кк_дії = контекст.кк_дії
      },
      дані_секції.тіло,
      місцезнаходження,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    вихід::вміст = обʼєкт_секції;
    вернути так;
  }

  дія скомпілювати_визначити_секцію(
    компілятор: *Компілятор,
    контекст: Контекст,
    дані_секції: розбирач::ДаніСполукиСекція,
    місцезнаходження: розбирач::Місцезнаходження,
    вихід: **ОбʼєктСекції,
    вихід_помилки: **Помилка,
  ) -> логічне {
    змінна успіх = ні;

    змінна назва_кд = кд { пусто, 0 };

    якщо не розбирач::отримати_значення_слова(
      контекст.розібраний_текст.текст_коду.значення,
      контекст.розібраний_текст.слова.дані[дані_секції.ідентифікатор.значення.позиція_слова],
      назва_кд::адреса
    ) {
      вихід_помилки::вміст = створити_помилку(
        компілятор,
        "Не вдалось отримати назву секції.",
        створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
      );
      вернути ні;
    }

    змінна назва = отримати_назву(компілятор, назва_кд);

    змінна обʼєкт_з_середовища: Обʼєкт;
    якщо знайти_в_середовищі(контекст.середовище, назва, обʼєкт_з_середовища::адреса) {
      якщо обʼєкт_з_середовища.вид == ВидОбʼєктаСекція {
        змінна обʼєкт_секції = обʼєкт_з_середовища.дані як *ОбʼєктСекції;

        успіх = скомпілювати_тіло(
          компілятор,
          Контекст {
            розібраний_текст = контекст.розібраний_текст,
            середовище = обʼєкт_секції.середовище,
            кк_дії = контекст.кк_дії
          },
          дані_секції.тіло,
          місцезнаходження,
          вихід_помилки
        );
        якщо не успіх {
          вернути ні;
        }

        вихід::вміст = обʼєкт_секції;
        вернути так;
      }

      вихід_помилки::вміст =створити_помилку(
        компілятор,
        "Субʼєкт вже визначено.",
        створити_місцезнаходження_помилки_з_позиції_слова(компілятор, контекст.розібраний_текст, дані_секції.ідентифікатор.значення.позиція_слова)
      );
      вернути ні;
    }

    успіх = скомпілювати_секцію(
      компілятор,
      контекст,
      пусто,
      дані_секції,
      місцезнаходження,
      вихід,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    змінити_в_середовищі(
      контекст.середовище,
      назва,
      Обʼєкт { ВидОбʼєктаСекція, вихід::вміст }
    );

    вернути так;
  }
}
