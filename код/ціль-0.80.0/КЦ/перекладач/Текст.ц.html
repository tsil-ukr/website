взяти визначення КЦ;

простір КЦ {
  дія скомпілювати_текст(компілятор: *Компілятор, контекст: Контекст, дані_тексту: розбирач::ДаніСполукиТекст, місцезнаходження: розбирач::Місцезнаходження, вихід: *Обʼєкт, вихід_помилки: **Помилка) -> логічне {
    змінна значення_кд = кд { пусто, 0 };

    якщо не розбирач::отримати_значення_слова(
      контекст.розібраний_текст.текст_коду.значення,
      контекст.розібраний_текст.слова.дані[дані_тексту.позиція_слова],
      значення_кд::адреса
    ) {
      вихід_помилки::вміст = створити_помилку(
        компілятор,
        "Не вдалось отримати текст.",
        створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
      );
      вернути ні;
    }
    значення_кд.розмір -= 2;
    значення_кд.дані = значення_кд.дані[1]::адреса;

    змінна тип_значення = отримати_тип_кд(компілятор);
    змінна розмір_значення: природне = 0;
    змінна дані_значення: *п8 = пусто;

    якщо дані_тексту.префікс.заповнено {
      змінна префікс_кд = кд { пусто, 0 };

      якщо не розбирач::отримати_значення_слова(
        контекст.розібраний_текст.текст_коду.значення,
        контекст.розібраний_текст.слова.дані[дані_тексту.префікс.значення.позиція_слова],
        префікс_кд::адреса
      ) {
        вихід_помилки::вміст = створити_помилку(
          компілятор,
          "Не вдалось отримати префікс тексту.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        );
        вернути ні;
      }

      якщо біб::перевірити_чи_кд_рівні(префікс_кд, "кд") або біб::перевірити_чи_кд_рівні(префікс_кд, "т8") {
        змінна оброблене_значення_кд = кд { пусто, 0 };
        біб::обробити_екранізації_в_кд(
          значення_кд,
          так,
          оброблене_значення_кд::адреса,
          пусто,
          пусто
        );
        розмір_значення = оброблене_значення_кд.розмір;
        дані_значення = оброблене_значення_кд.дані;
      } інакше якщо біб::перевірити_чи_кд_рівні(префікс_кд, "ю8") {
        змінна значення_ю8 = ю8 { пусто, 0 };

        якщо не біб::перекодувати_в_ю8(значення_кд, так, значення_ю8::адреса, пусто, пусто) {
          вихід_помилки::вміст = створити_помилку(
            компілятор,
            "Неможливо створити текст.",
            створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
          );
          вернути ні;
        }

        змінна оброблене_значення_ю8 = ю8 { пусто, 0 };
        біб::обробити_екранізації_в_ю8(
          значення_ю8,
          так,
          оброблене_значення_ю8::адреса,
          пусто,
          пусто
        );
        біб::звільнити(значення_ю8.дані);
        тип_значення = отримати_тип_ю8(компілятор);
        розмір_значення = оброблене_значення_ю8.розмір;
        дані_значення = оброблене_значення_ю8.дані;
      } інакше {
        вихід_помилки::вміст = створити_помилку(
          компілятор,
          "Неможливо створити текст.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        );
        вернути ні;
      }
    } інакше {
      змінна оброблене_значення_кд = кд { пусто, 0 };
      біб::обробити_екранізації_в_кд(
        значення_кд,
        так,
        оброблене_значення_кд::адреса,
        пусто,
        пусто
      );
      розмір_значення = оброблене_значення_кд.розмір;
      дані_значення = оброблене_значення_кд.дані;
    }

    змінна пм_значення_розміру = ПМ::створити_п64(компілятор.пм_модуль, розмір_значення);
    змінна пм_значення_даних: *ПМ::Значення = пусто;

    пм_значення_даних = ПМ::створити_константу_даних(компілятор.пм_модуль, розмір_значення + 1, дані_значення);

    змінна аргументи = біб::виділити<*ПМ::Значення>(2);
    аргументи[0] = пм_значення_даних;
    аргументи[1] = пм_значення_розміру;

    змінна пм_значення = ПМ::створити_обʼєкт(
      компілятор.пм_модуль,
      тип_значення.пм_тип,
      2,
      аргументи
    );

    вихід::вміст = Обʼєкт { ВидОбʼєктаЗначення, ОбʼєктЗначення { тип_значення, пм_значення } };
    вернути так;
  }
