  дія скомпілювати_параметри_шаблону(
    компілятор: *Компілятор,
    контекст: Контекст,
    параметри_розбирача: розбирач::ПараметриШаблону,
    місцезнаходження: розбирач::Місцезнаходження,
    вихід: *Параметри,
    вихід_помилки: **Помилка,
  ) -> логічне {
    змінна успіх = ні;
    змінна параметри = Параметри { 0, пусто, 0 };

    змінна п: природне = 0;

    поки п < параметри_розбирача.розмір {
      змінна параметр_розбирача = параметри_розбирача.дані[п];

      змінна назва_кд = кд { пусто, 0 };

      якщо не розбирач::отримати_значення_слова(
        контекст.розібраний_текст.текст_коду.значення,
        контекст.розібраний_текст.слова.дані[параметр_розбирача.ідентифікатор.позиція_слова],
        назва_кд::адреса
      ) {
        знищити_параметри(компілятор, параметри);

        вихід_помилки::вміст = створити_помилку(
          компілятор,
          "Не вдалось отримати назву параметра шаблону.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        );
        вернути ні;
      }

      змінна назва = отримати_назву(компілятор, назва_кд);

      якщо параметр_розбирача.тип == пусто {
        покласти_параметр(
          компілятор,
          параметри::адреса,
          Параметр {
            назва,
            пусто,
            Місцезнаходження { контекст.розібраний_текст, параметр_розбирача.ідентифікатор.позиція_слова },
            Місцезнаходження { пусто, 0 }
          }
        );
      } інакше {
        змінна от: *ОбʼєктТипу;
        успіх = скомпілювати_значення_як_тип(
          компілятор,
          контекст,
          параметр_розбирача.тип,
          так,
          от::адреса,
          вихід_помилки
        );
        якщо не успіх {
          знищити_параметри(компілятор, параметри);
          вернути ні;
        }

        покласти_параметр(
          компілятор,
          параметри::адреса,
          Параметр {
            назва,
            от,
            Місцезнаходження { контекст.розібраний_текст, параметр_розбирача.ідентифікатор.позиція_слова },
            Місцезнаходження { контекст.розібраний_текст, параметр_розбирача.тип.місцезнаходження.позиція_початкового_слова }
          }
        );
      }

      п += 1;
    }

    вихід::вміст = параметри;
    вернути так;
  }

      дані_шаблону,
      місцезнаходження,
      вихід_помилки
    );

    якщо не успіх {
      вернути ні;
    }

    якщо обʼєкт_задекларованого_шаблону.сполука.вид == розбирач::ВидСполукиСтруктура {
      якщо дані_шаблону.значення.вид == розбирач::ВидСполукиСтруктура {
        вернути дозаповнити_задекларований_шаблон_структури(
          компілятор,
          контекст,
          обʼєкт_задекларованого_шаблону,
          дані_шаблону,
          місцезнаходження,
          вихід_помилки
        );
      }
    } інакше якщо обʼєкт_задекларованого_шаблону.сполука.вид == розбирач::ВидСполукиДія {
      якщо дані_шаблону.значення.вид == розбирач::ВидСполукиДія {
        вернути дозаповнити_задекларований_шаблон_дії(
          компілятор,
          контекст,
          обʼєкт_задекларованого_шаблону,
          дані_шаблону,
          місцезнаходження,
          вихід_помилки
        );
      }
    }

    вихід_помилки::вміст = створити_помилку_субʼєкт_вже_визначено(
      компілятор,
      Місцезнаходження { контекст.розібраний_текст, дані_шаблону.ідентифікатор.значення.позиція_слова },
      назва_кд
    );
    вернути ні;
  }

  дія скомпілювати_шаблон(
    компілятор: *Компілятор,
    контекст: Контекст,
    дані_шаблону: розбирач::ДаніСполукиШаблон,
    місцезнаходження: розбирач::Місцезнаходження,
    вихід: **ОбʼєктШаблону,
    вихід_помилки: **Помилка,
  ) -> логічне {
    змінна успіх = ні;
    змінна параметри: Параметри;

    змінна назва_кд = кд { пусто, 0 };

    якщо дані_шаблону.ідентифікатор.заповнено {
      якщо не розбирач::отримати_значення_слова(
        контекст.розібраний_текст.текст_коду.значення,
        контекст.розібраний_текст.слова.дані[дані_шаблону.ідентифікатор.значення.позиція_слова],
        назва_кд::адреса
      ) {
        вихід_помилки::вміст = створити_помилку(
          компілятор,
          "Не вдалось отримати назву шаблону.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        );
        вернути ні;
      }
    }

    змінна назва = отримати_назву(компілятор, назва_кд);

    успіх = скомпілювати_параметри_шаблону(
      компілятор,
      контекст,
      дані_шаблону.параметри,
      місцезнаходження,
      параметри::адреса,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    якщо дані_шаблону.значення.вид == розбирач::ВидСполукиДія {
      змінна дані_дії = дані_шаблону.значення.дані як розбирач::ДаніСполукиДія;
      якщо дані_дії.видимість != розбирач::ВИДИМІСТЬ_ВНУТРІШНЯ {
        вихід_помилки::вміст = створити_помилку(
          компілятор,
          "Дія шаблону повинна мати внутрішню видимість.",
          створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
        );
        вернути ні;
      }
    }

    змінна обʼєкт_шаблону = виділити_обʼєкт_шаблону(компілятор);

    обʼєкт_шаблону.назва = назва;
    обʼєкт_шаблону.параметри = параметри;
    обʼєкт_шаблону.сполука = дані_шаблону.значення;
    обʼєкт_шаблону.контекст_компіляції = контекст;

    вихід::вміст = обʼєкт_шаблону;
    вернути так;
  }

  дія скомпілювати_визначити_шаблон(
    компілятор: *Компілятор,
    контекст: Контекст,
    дані_шаблону: розбирач::ДаніСполукиШаблон,
    місцезнаходження: розбирач::Місцезнаходження,
    вихід: **ОбʼєктШаблону,
    вихід_помилки: **Помилка,
  ) -> логічне {
    змінна успіх = ні;
    змінна помилка: *Помилка = пусто;
    змінна назва_кд = кд { пусто, 0 };

    якщо не розбирач::отримати_значення_слова(
      контекст.розібраний_текст.текст_коду.значення,
      контекст.розібраний_текст.слова.дані[дані_шаблону.ідентифікатор.значення.позиція_слова],
      назва_кд::адреса
    ) {
      вихід_помилки::вміст = створити_помилку(
        компілятор,
        "Не вдалось отримати назву шаблону.",
        створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
      );
      вернути ні;
    }

    змінна назва = отримати_назву(компілятор, назва_кд);

    змінна обʼєкт_з_середовища: Обʼєкт;
    якщо знайти_в_середовищі(контекст.середовище, назва, обʼєкт_з_середовища::адреса) {
      якщо обʼєкт_з_середовища.вид == ВидОбʼєктаШаблон {
        змінна обʼєкт_шаблону = обʼєкт_з_середовища.дані як *ОбʼєктШаблону;

        успіх = дозаповнити_задекларований_шаблон(
          компілятор,
          контекст,
          обʼєкт_шаблону,
          дані_шаблону,
          місцезнаходження,
          помилка::адреса
        );

        якщо не успіх {
          вихід_помилки::вміст = помилка;
          вернути ні;
        }

        вихід::вміст = обʼєкт_шаблону;
        вернути так;
      }

      вихід_помилки::вміст = створити_помилку_субʼєкт_вже_визначено(
        компілятор,
        Місцезнаходження { контекст.розібраний_текст, дані_шаблону.ідентифікатор.значення.позиція_слова },
        назва_кд
      );
      вернути ні;
    }

    успіх = скомпілювати_шаблон(
      компілятор,
      контекст,
      дані_шаблону,
      місцезнаходження,
      вихід,
      вихід_помилки
    );
    якщо не успіх {
      вернути ні;
    }

    змінити_в_середовищі(
      контекст.середовище,
      назва,
      Обʼєкт { ВидОбʼєктаШаблон, вихід::вміст }
    );

    вернути так;
  }
