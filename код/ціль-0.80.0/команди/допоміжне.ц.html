внутрішня дія отримати_батьків(
  шлях: ю8,
  глибина: природне,
  вихід: *ю8
) -> логічне {
  змінна шлях_до_папки = шлях;

  змінна глиб = природне(0);

  поки шлях_до_папки.розмір > 0 {
    якщо шлях_до_папки.дані[шлях_до_папки.розмір - 1] == 47 { // '/'
      якщо глиб == глибина {
        вихід::вміст = шлях_до_папки;
        вернути так;
      } інакше {
        глиб += 1;
      }
    }

    шлях_до_папки.розмір -= 1;
  }

  вернути ні;
}

дія зробити_шлях_до_основи(
  шлях_до_цілі: ю8,
  вихід: *ю8
) -> логічне {
  змінна результат: ю8;

  якщо не отримати_батьків(шлях_до_цілі, 1, результат::адреса) {
    вернути ні;
  }

  змінна доповнення = ю8"основа.в.ц";

  результат.дані = біб::виділити<п8>(результат.розмір + доповнення.розмір);
  змінна п: природне = 0;
  поки п < результат.розмір {
    результат.дані[п] = шлях_до_цілі.дані[п];
    п += 1;
  }
  п = 0;
  поки п < доповнення.розмір {
    результат.дані[результат.розмір + п] = доповнення.дані[п];
    п += 1;
  }
  результат.розмір += доповнення.розмір;

  вихід::вміст = результат;

  вернути так;
}
