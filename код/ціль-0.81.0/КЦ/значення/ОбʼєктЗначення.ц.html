        ВидОбʼєктаЗначення,
        обʼєкт_значення_для_результату

      вернути так;

      створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)

  ) -> логічне {
    якщо аргумент.обʼєкт.вид != ВидОбʼєктаТип {
      вихід_помилки::вміст = створити_помилку(
        компілятор,
        "Не вдалось скомпілювати",
        створити_місцезнаходження_помилки_з_місцезнаходження_розбирача(компілятор, контекст.розібраний_текст, місцезнаходження)
      );
      вернути ні;
    }

    змінна новий_тип = аргумент.обʼєкт.дані як *ОбʼєктТипу;
    якщо новий_тип == обʼєкт_значення.тип {
      вихід::вміст = Обʼєкт { ВидОбʼєктаЗначення, дані = обʼєкт_значення };
        вернути так;
    }

    якщо обʼєкт_значення.тип.вид == ВидОбʼєктаТипуАдреса {
      якщо новий_тип.вид == ВидОбʼєктаТипуАдреса {
        обʼєкт_значення.тип = новий_тип;
        вихід::вміст = Обʼєкт { ВидОбʼєктаЗначення, дані = обʼєкт_значення };
        вернути так;
      }
      якщо новий_тип.вид == ВидОбʼєктаТипуДія {
        // потім: перевірити чи це тип невідомої адреси
        обʼєкт_значення.тип = новий_тип;
        вихід::вміст = Обʼєкт { ВидОбʼєктаЗначення, дані = обʼєкт_значення };
        вернути так;
      }
    }

    якщо обʼєкт_значення.тип.вид == ВидОбʼєктаТипуДія {
      // потім: перевірити тут тоже!!!
      якщо новий_тип.вид == ВидОбʼєктаТипуАдреса {
        обʼєкт_значення.тип = новий_тип;
        вихід::вміст = Обʼєкт { ВидОбʼєктаЗначення, дані = обʼєкт_значення };
        вернути так;
      }
    }

    якщо обʼєкт_значення.тип.вид == ВидОбʼєктаТипуВаріація {
      змінна дані_варіації = обʼєкт_значення.тип.дані як ДаніОбʼєктаТипуВаріація;
      змінна знайдений_тип_варіації: *ОбʼєктТипу = пусто;

          знайдений_тип_варіації = дані_варіації.типи[п];
        }
        п += 1;
      }
      якщо знайдений_тип_варіації == пусто {
        вихід_помилки::вміст = створити_помилку(

        );
        вернути ні;

      якщо дані_варіації.найбільший_тип == компілятор.тип_р64 і знайдений_тип_варіації.вид == ВидОбʼєктаТипуАдреса {

      } інакше {

      }

      обʼєкт_значення.тип = знайдений_тип_варіації;
