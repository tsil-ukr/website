взяти біб Памʼять;
взяти біб Накопичувач;
взяти біб Кодування;

простір СховищеНазв {
  структура Назва {
    значення: кд;
    значення_ю8: ю8;
    має_значення_ю8: логічне;

    внутрішня дія як_ю8(я: *Я) -> ю8 {
      якщо не я.має_значення_ю8 {
        біб::перекодувати_в_ю8(
          я.значення,
          так,
          я.значення_ю8::адреса,
          пусто,
          пусто
        );

        я.має_значення_ю8 = так;
      }

      вернути я.значення_ю8;
    }
  }

  структура Сховище {
    назви: біб::Накопичувач<*Назва>;

    внутрішня дія отримати(я: *Я, значення: кд) -> *Назва {
      змінна п: природне = 0;

      поки п < я.назви.розмір {
        якщо я.назви.дані[п].значення.рівн(значення) {
          вернути я.назви.дані[п];
        }

        п += 1;
      }

      змінна нова_назва = біб::виділити<Назва>(1);

      нова_назва.значення = кд {
        біб::виділити<п8>(значення.розмір),
        значення.розмір,
      };

      нова_назва.значення_ю8 = ю8 { пусто, 0 };
      нова_назва.має_значення_ю8 = ні;

      п = 0;
      поки п < значення.розмір {
        нова_назва.значення.дані[п] = значення.дані[п];
        п += 1;
      }

      біб::покласти_в_накопичувач<*Назва>(
        я.назви::адреса,
        нова_назва
      );

      вернути нова_назва;
    }
  }


  внутрішня дія створити_сховище() -> *Сховище {
    змінна сховище = біб::виділити<Сховище>(1);

    сховище.назви = біб::зробити_накопичувач<*Назва>(0);

    вернути сховище;
  }

  внутрішня дія знищити_сховище(сховище: *Сховище) {
    змінна п: природне = 0;

    поки п < сховище.назви.розмір {
      біб::звільнити(сховище.назви.дані[п].значення.дані);
      біб::звільнити(сховище.назви.дані[п]);

      п += 1;
    }

    біб::звільнити(сховище.назви.дані);
    біб::звільнити(сховище);
  }
}
