    змінна п: природне = 0;

      п += 1;
    }
  }

  внутрішня дія покласти_кд_п64(нк: *біб::Накопичувач<п8>, значення: п64) {
    змінна значення_ю8 = ю8 { пусто, 0 };

    біб::перекодувати_з_ю8(значення_ю8, так, так, значення_кд::адреса, пусто, позиція_помилки::адреса);
    біб::звільнити(значення_ю8.дані);

    покласти_кд(нк, значення_кд);

    структури: біб::Накопичувач<Структура>;
    глобальні: біб::Накопичувач<ю8>;
    функції: біб::Накопичувач<*Функція>;

      змінна нн = біб::зробити_накопичувач<п8>(0);
      покласти_кд(нн::адреса, "назва_");
      покласти_кд_п64(нн::адреса, п64(я.лн));

      змінна назва = я.сн.отримати(кд { нн.дані, нн.розмір });

      біб::звільнити(нн.дані);

      я.лн += 1;

      вернути назва;
    }

    внутрішня дія отримати_сіанг(я: *Я, вихід_даних: **п8, вихід_розміру: *природне) {
      змінна нк = біб::зробити_накопичувач<п8>(0);
      змінна п: природне;

      змінна цільінт_дані: *п8;
      змінна цільінт_розмір = отримати_цільінт(цільінт_дані::адреса);

      п = 0;
      поки п < цільінт_розмір {
        біб::покласти_в_накопичувач<п8>(нк::адреса, цільінт_дані[п]);

        п += 1;
      }

      п = 0;
      поки п < я.структури.розмір {
        покласти_ю8(нк::адреса, я.СІАНГ_ТАЙПДЕФ);
        покласти_ю8(нк::адреса, ю8" ");
        покласти_ю8(нк::адреса, я.СІАНГ_СТРАКТ);
        покласти_ю8(нк::адреса, ю8" ");
        покласти_ю8(нк::адреса, я.структури.дані[п].назва.як_ю8());
        покласти_ю8(нк::адреса, ю8" ");
        покласти_ю8(нк::адреса, я.структури.дані[п].назва.як_ю8());
        покласти_ю8(нк::адреса, ю8";\р");

        п += 1;
      }

      п = 0;
      поки п < я.типи.розмір {
        покласти_ю8(нк::адреса, я.типи.дані[п]);
        покласти_ю8(нк::адреса, ю8"\р");

        п += 1;
      }

      п = 0;
      поки п < я.структури.розмір {
        покласти_ю8(нк::адреса, я.структури.дані[п].код);
        покласти_ю8(нк::адреса, ю8"\р");

        п += 1;
      }

      п = 0;
      поки п < я.глобальні.розмір {
        покласти_ю8(нк::адреса, я.глобальні.дані[п]);
        покласти_ю8(нк::адреса, ю8"\р");

        п += 1;
      }

      п = 0;
      поки п < я.функції.розмір {
        покласти_ю8(нк::адреса, я.функції.дані[п].оголошення);
        покласти_ю8(нк::адреса, ю8";\р");

        п += 1;
      }

      п = 0;
      поки п < я.функції.розмір {
        змінна ф = я.функції.дані[п];
        змінна пп: природне = 0;

        якщо ф.кроки.розмір > 0 {
          покласти_ю8(нк::адреса, ф.оголошення);
          покласти_ю8(нк::адреса, ю8" {\р");
          покласти_ю8(нк::адреса, ф.аргументи);

          поки пп < ф.кроки.розмір {
            змінна крок = ф.кроки.дані[пп];

            покласти_ю8(нк::адреса, крок.код);

            пп += 1;
          }

          покласти_ю8(нк::адреса, ю8"}\р");
        }

        п += 1;
      }

      вихід_даних::вміст = нк.дані;
      вихід_розміру::вміст = нк.розмір;
    }
  }

    покласти_ю8(нктф::адреса, тип_результату.як_ю8());
    покласти_ю8(нктф::адреса, ю8" (*)(");

    змінна п: природне = 0;
    поки п < дані_типу_інструкція.параметри.розмір {

        покласти_ю8(нктф::адреса, ю8", ");
      }

      п += 1;
    }

    покласти_ю8(нктф::адреса, ю8")");

    вернути ю8 { нктф.дані, нктф.розмір };
  }

  внутрішня дія скомпілювати_тип(файл: *Файл, тип: *ПМ::Тип, для_виконати: логічне) -> *СховищеНазв::Назва {
    змінна модуль = тип.модуль;

    якщо тип.дані_компіляції == пусто {
      якщо тип == модуль.тип_ніщо {
        тип.дані_компіляції = модуль.сн.отримати("ніщо");
      } інакше якщо тип == модуль.тип_адреса {
        тип.дані_компіляції = модуль.сн.отримати("адреса");
      } інакше якщо тип == модуль.тип_логічне {
        тип.дані_компіляції = модуль.сн.отримати("логічне");
      } інакше якщо тип == модуль.тип_ц8 {
        тип.дані_компіляції = модуль.сн.отримати("п8");
      } інакше якщо тип == модуль.тип_ц16 {
        тип.дані_компіляції = модуль.сн.отримати("п16");
      } інакше якщо тип == модуль.тип_ц32 {
        тип.дані_компіляції = модуль.сн.отримати("п32");
      } інакше якщо тип == модуль.тип_ц64 {
        тип.дані_компіляції = модуль.сн.отримати("п64");
      } інакше якщо тип == модуль.тип_р32 {
        тип.дані_компіляції = модуль.сн.отримати("р32");
      } інакше якщо тип == модуль.тип_р64 {
        тип.дані_компіляції = модуль.сн.отримати("р64");
      } інакше якщо тип.вид == ПМ::ВидТипуІнструкція {
        змінна дані_типу_інструкція = тип.дані як ПМ::ДаніТипуІнструкція;

        змінна назва = файл.онн();

        змінна нктф = біб::зробити_накопичувач<п8>(0);

        змінна тип_результату = скомпілювати_тип(файл, дані_типу_інструкція.тип_результату, ні);

        покласти_ю8(нктф::адреса, файл.СІАНГ_ТАЙПДЕФ);
        покласти_ю8(нктф::адреса, ю8" ");
        покласти_ю8(нктф::адреса, тип_результату.як_ю8());
        покласти_ю8(нктф::адреса, ю8" (*");
        покласти_ю8(нктф::адреса, назва.як_ю8());
        покласти_ю8(нктф::адреса, ю8")(");

        змінна п: природне = 0;
        поки п < дані_типу_інструкція.параметри.розмір {
          змінна параметр = дані_типу_інструкція.параметри.дані[п];

          змінна тип = скомпілювати_тип(файл, параметр.тип, ні);

          покласти_ю8(нктф::адреса, тип.як_ю8());

          якщо п != (дані_типу_інструкція.параметри.розмір - 1) {
            покласти_ю8(нктф::адреса, ю8", ");
          }

          п += 1;
        }

        покласти_ю8(нктф::адреса, ю8");");

        біб::покласти_в_накопичувач<ю8>(
          файл.типи::адреса,
          ю8 { нктф.дані, нктф.розмір }
        );

        тип.дані_компіляції = файл.сн.отримати("адреса");

        якщо для_виконати {
          вернути назва;
        }
      } інакше якщо тип.вид == ПМ::ВидТипуСтруктура {
        змінна дані_типу_структура = тип.дані як ПМ::ДаніТипуСтруктура;

        змінна нкс = біб::зробити_накопичувач<п8>(0);
        покласти_ю8(нкс::адреса, файл.СІАНГ_СТРАКТ);
        покласти_ю8(нкс::адреса, ю8" ");
        покласти_ю8(нкс::адреса, дані_типу_структура.назва.як_ю8());
        покласти_ю8(нкс::адреса, ю8" {\р");

        змінна п: природне = 0;
        поки п < дані_типу_структура.параметри.розмір {
          змінна параметр = дані_типу_структура.параметри.дані[п];

          змінна тип = скомпілювати_тип(файл, параметр.тип, ні);
          змінна назва = параметр.назва;

          покласти_ю8(нкс::адреса, ю8"  ");
          покласти_ю8(нкс::адреса, тип.як_ю8());
          покласти_ю8(нкс::адреса, ю8" ");
          покласти_ю8(нкс::адреса, назва.як_ю8());
          покласти_ю8(нкс::адреса, ю8";\р");

          п += 1;
        }

        покласти_ю8(нкс::адреса, ю8"};");

        змінна с: Структура;
        с.назва = дані_типу_структура.назва;
        с.код = ю8 { нкс.дані, нкс.розмір };

        біб::покласти_в_накопичувач<Структура>(файл.структури::адреса, с);

        тип.дані_компіляції = дані_типу_структура.назва;
      } інакше якщо тип.вид == ПМ::ВидТипуМасив {
      }
    }

    вернути тип.дані_компіляції як *СховищеНазв::Назва;
  }

  внутрішня дія скомпілювати_крок(модуль: *ПМ::Модуль, файл: *Файл, функція: *Функція, крок: *ПМ::Крок) -> *Крок {
    якщо крок.дані_компіляції == пусто {
      крок.назва = файл.онн();

      змінна к = функція.створити_крок(крок.назва);
      змінна перший_крок = функція.кроки.розмір > 1;

      крок.дані_компіляції = к;

      змінна нк = біб::зробити_накопичувач<п8>(0);

      якщо перший_крок {
        // покласти_ю8(нк::адреса, ю8"{\р");
      }
      покласти_ю8(нк::адреса, крок.назва.як_ю8());
      покласти_ю8(нк::адреса, ю8": ;\р");

      змінна пп: природне = 0;
      поки пп < крок.вказівки.розмір {
        змінна вказівка = крок.вказівки.дані[пп];

        скомпілювати_значення(файл, функція, вказівка, нк::адреса);

        пп += 1;
      }

      якщо перший_крок {
        // покласти_ю8(нк::адреса, ю8"}\р");
      }

      к.код = ю8 { нк.дані, нк.розмір };
    }

    вернути крок.дані_компіляції як *Крок;
